*&---------------------------------------------------------------------*
*& Form format_program_lines
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*& -->  p1        text
*& <--  p2        text
*&---------------------------------------------------------------------*
FORM format_program_lines .
  LOOP AT gt_program_code INTO gs_program_code.
    IF gs_program_code-line+0(1) = '*'.
      CONTINUE.
    ENDIF.
    SPLIT gs_program_code-line AT '"' INTO gs_program_code-line DATA(lv_dummy).
    CONDENSE gs_program_code-line.
    CONCATENATE lv_program_lines gs_program_code-line INTO lv_program_lines.

    PERFORM escape_json_string CHANGING  gs_program_code-line.
  ENDLOOP.
ENDFORM.

*&---------------------------------------------------------------------*
*& Form extract_cds_code
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*&      --> P_PRG
*&---------------------------------------------------------------------*
FORM extract_cds_code  USING    p_p_prg TYPE progname.

  DATA: lv_code    TYPE string,
        lv_offset  TYPE i VALUE 0,
        lv_len     TYPE i,
        lv_line    TYPE abaptxt255,
        lv_cleaned TYPE string,
        lv_pos     TYPE i.

  CLEAR gt_program_code.

  " Read full CDS source
  SELECT SINGLE source INTO @lv_code
    FROM ddddlsrc
    WHERE ddlname = @p_p_prg.

  IF sy-subrc = 0 AND lv_code IS NOT INITIAL.
    " Remove technical metadata (everything from /*+[internal] ... onwards)
    FIND '/*+[' IN lv_code MATCH OFFSET lv_pos.
    IF sy-subrc = 0.
      lv_cleaned = lv_code+0(lv_pos). " Extract only valid code before metadata
    ELSE.
      lv_cleaned = lv_code.
    ENDIF.

    " Break into 255-character lines
    WHILE lv_offset < strlen( lv_cleaned ).
      lv_len = strlen( lv_cleaned ) - lv_offset.
      IF lv_len > 255.
        lv_len = 255.
      ENDIF.
      lv_line = lv_cleaned+lv_offset(lv_len).
      APPEND lv_line TO gt_program_code.
      ADD 255 TO lv_offset.
    ENDWHILE.

  ELSE.
    MESSAGE |{ p_p_prg } not found in DDDLSRC| TYPE 'S'.
  ENDIF.

ENDFORM.

*& Form show_token_details
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*& -->  p1        text
*& <--  p2        text
*&---------------------------------------------------------------------*
FORM show_token_details
  USING    iv_json          TYPE cstring
  CHANGING ev_token_details TYPE string.
  TYPES: BEGIN OF ty_usage,
           prompttokencount     TYPE i,
           thoughtstokencount   TYPE i,
           candidatestokencount TYPE i,
           totaltokencount      TYPE i,
         END OF ty_usage.

  TYPES: BEGIN OF ty_response,
           usagemetadata TYPE ty_usage,
         END OF ty_response.
  DATA: ls_response TYPE ty_response.

  TRY.
      CALL METHOD /ui2/cl_json=>deserialize
        EXPORTING
          json = iv_json
        CHANGING
          data = ls_response.

      ev_token_details = |--- Token Usage Details ---| &&
                            cl_abap_char_utilities=>newline &&
                            |Prompt Token Count     : { ls_response-usagemetadata-prompttokencount }| &&
                            cl_abap_char_utilities=>newline &&
                            |Thoughts Token Count   : { ls_response-usagemetadata-thoughtstokencount }| &&
                            cl_abap_char_utilities=>newline &&
                            |Candidates Token Count : { ls_response-usagemetadata-candidatestokencount }| &&
                            cl_abap_char_utilities=>newline &&
                            |Total Token Count      : { ls_response-usagemetadata-totaltokencount }|.

    CATCH cx_root INTO DATA(lx_json).
      ev_token_details = |Token parsing failed: { lx_json->get_text( ) }|.
  ENDTRY.
ENDFORM.

FORM escape_json_string CHANGING cv_output TYPE text255.

  " Order matters: escape the escape character (backslash) first
  REPLACE ALL OCCURRENCES OF '\' IN cv_output WITH '\\'.

  " Escape double quotes
  REPLACE ALL OCCURRENCES OF '"' IN cv_output WITH '\"'.

  " Escape control characters (New Line, Carriage Return, Tabs)
  " Note: CL_ABAP_CHAR_UTILITIES provides the hex values for these
  REPLACE ALL OCCURRENCES OF cl_abap_char_utilities=>newline        IN cv_output WITH '\n'.
  REPLACE ALL OCCURRENCES OF cl_abap_char_utilities=>cr_lf          IN cv_output WITH '\r\n'.
  REPLACE ALL OCCURRENCES OF cl_abap_char_utilities=>horizontal_tab IN cv_output WITH '\t'.

ENDFORM.

*& Form get_json

*& text
*&---------------------------------------------------------------------*
*&      --> LV_PROMPT
*&      --> LV_SYS_INSTRUCT
*&      <-- LV_PAYLOAD
*&---------------------------------------------------------------------*
FORM get_json  USING    pv_prompt
                        pv_sys_instruct
               CHANGING pv_payload.

* 1. Define the nested structure matching your Gemini payload
  TYPES: BEGIN OF ty_part,
           text TYPE string,
         END OF ty_part,
         tt_parts TYPE STANDARD TABLE OF ty_part WITH EMPTY KEY.

  TYPES: BEGIN OF ty_content,
           parts TYPE tt_parts,
           role  TYPE string, " Optional for Gemini
         END OF ty_content,
         tt_contents TYPE STANDARD TABLE OF ty_content WITH EMPTY KEY.

  TYPES: BEGIN OF ty_gen_config,
           stop_sequences TYPE string_table, " stopSequences in JSON
           temperature    TYPE f,            " Maps to 0 or decimals
         END OF ty_gen_config.

  TYPES: BEGIN OF ty_system_instruction,
           parts TYPE tt_parts,
         END OF ty_system_instruction.

  TYPES: BEGIN OF ty_gemini_payload,
           system_instruction TYPE ty_system_instruction, " systemInstruction
           contents           TYPE tt_contents,
           generation_config  TYPE ty_gen_config,        " generationConfig
         END OF ty_gemini_payload.

  DATA: ls_payload TYPE ty_gemini_payload,
        lv_json    TYPE string.

* ---------------------------------------------------------
* 2. FILL INTERNAL STRUCTURE (ABAP -> JSON)
* ---------------------------------------------------------
  ls_payload-system_instruction-parts = VALUE #( ( text = pv_sys_instruct ) ).
  ls_payload-contents = VALUE #( ( parts = VALUE #( ( text = pv_prompt ) ) ) ).
  ls_payload-generation_config = VALUE #(
      stop_sequences = VALUE #( ( `Title` ) )
      temperature    = '0.0'
  ).

* Serialize: Convert ABAP structure to CamelCase JSON
  pv_payload = /ui2/cl_json=>serialize(
              data        = ls_payload
              pretty_name = /ui2/cl_json=>pretty_mode-camel_case ).

ENDFORM.
*&---------------------------------------------------------------------*
*& Form calculate_results
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*&      --> LV_RESULT_JSON
*&      --> GV_TOKEN_INFO
*&      <-- LI_HTML_FILE
*&---------------------------------------------------------------------*
FORM calculate_results  USING    p_lv_result_json
                                 p_gv_token_info
                        CHANGING pi_html_file TYPE gty_tt_file.

  DATA: ls_html_file TYPE LINE OF gty_tt_file.

  /ui2/cl_json=>deserialize(
    EXPORTING
        json = lv_result_json
        pretty_name = /ui2/cl_json=>pretty_mode-camel_case
    CHANGING
        data = lr_data ).

  CHECK lr_data IS BOUND.
  ASSIGN lr_data->* TO FIELD-SYMBOL(<data>).
  CHECK <data> IS ASSIGNED.

  ASSIGN COMPONENT 'MODEL_VERSION' OF STRUCTURE <data> TO FIELD-SYMBOL(<lf_model>).
  CHECK <lf_model> IS ASSIGNED.
  ASSIGN <lf_model>->* TO <lv_model>.

  ASSIGN COMPONENT 'CANDIDATES' OF STRUCTURE <data> TO FIELD-SYMBOL(<lf_r_candidates>).
  CHECK <lf_r_candidates> IS ASSIGNED.
  ASSIGN <lf_r_candidates>->* TO <lt_candidates>.
  CHECK <lt_candidates> IS ASSIGNED.

  LOOP AT <lt_candidates> ASSIGNING FIELD-SYMBOL(<lf_candidates>).
    ASSIGN <lf_candidates>->* TO <data>.
    CHECK <data> IS ASSIGNED.
    ASSIGN COMPONENT 'CONTENT' OF STRUCTURE <data> TO FIELD-SYMBOL(<lf_r_content>).
    ASSIGN <lf_r_content>->* TO <lf_r_content>.

    CHECK <lf_r_content> IS ASSIGNED.
    ASSIGN COMPONENT 'PARTS' OF STRUCTURE <lf_r_content> TO FIELD-SYMBOL(<lf_ir_parts>).

    LOOP AT <lf_ir_parts>->* ASSIGNING FIELD-SYMBOL(<fs_parts>).
      ASSIGN COMPONENT 'TEXT' OF STRUCTURE <fs_parts>->* TO FIELD-SYMBOL(<lf_o_text>).
      ASSIGN <lf_o_text>->* TO FIELD-SYMBOL(<lf_text>).
      ls_html_file-line = <lf_text>.
      APPEND ls_html_file TO pi_html_file.
      CLEAR ls_html_file.
    ENDLOOP.
  ENDLOOP.

*  Appending the token details to the responce
  IF gv_token_info IS NOT INITIAL.
    ls_html_file-line =  |<div style="margin-top:20px; padding:10px; border:1px solid #ccc; background-color:#f9f9f9;">| &&
                       |<h3 style="color:#0072C6;">🔢 Token Usage Details</h3>| &&
                       |<pre style="white-space: pre-wrap;">{ gv_token_info }</pre>| &&
                       |</div>|.
    APPEND ls_html_file TO pi_html_file.
    CLEAR ls_html_file.

  ENDIF.

ENDFORM.
