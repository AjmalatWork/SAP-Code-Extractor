*&---------------------------------------------------------------------*
*& ALV event handler
*&---------------------------------------------------------------------*
*       Only reason this report uses a class at all: CL_GUI_ALV_GRID's
*       per-cell F4 (ONF4 event) requires an OO event handler - there is
*       no FORM-based callback for it. All other logic stays in FORMs.
*&---------------------------------------------------------------------*
CLASS lcl_alv_event_handler DEFINITION.
  PUBLIC SECTION.
    METHODS on_f4 FOR EVENT onf4 OF cl_gui_alv_grid
      IMPORTING e_fieldname es_row_no er_event_data.
    METHODS on_data_changed FOR EVENT data_changed OF cl_gui_alv_grid
      IMPORTING er_data_changed.
ENDCLASS.

" GO_ALV_EVENTS must be declared after the class it's typed as - moved
" here rather than into the Global Data block above for that reason.
DATA go_alv_events TYPE REF TO lcl_alv_event_handler.

CLASS lcl_alv_event_handler IMPLEMENTATION.
  METHOD on_f4.
    " Only the "Compare Against Version" column has a custom F4.
    CHECK e_fieldname = 'CMP_VERSNO'.

    FIELD-SYMBOLS: <fs_row>   TYPE ts_multi_obj,
                   <fs_field> TYPE any.

    READ TABLE gt_multi_obj ASSIGNING <fs_row> INDEX es_row_no-row_id.
    CHECK sy-subrc = 0.

    DATA: lv_object_type TYPE e071-object,
          lv_obj_name    TYPE vrsd-objname,
          lt_dummy       TYPE STANDARD TABLE OF vrsd,
          lt_versions    TYPE STANDARD TABLE OF vrsd,
          lt_f4_list     TYPE STANDARD TABLE OF ddshretval,
          ls_stable      TYPE lvc_s_stbl.

    lv_object_type = <fs_row>-object.
    IF lv_object_type = 'PROG'.
      lv_object_type = 'REPS'.
    ENDIF.

    " E071-OBJ_NAME (CHAR120) vs VRSD-OBJNAME (CHAR110) - same mismatch
    " as ENRICH_MULTI_OBJECTS, fixed the same way.
    lv_obj_name = <fs_row>-obj_name.

    " F4 intentionally shows ALL versions of this object in the version
    " database, not just the ones tied to the entered TRs.
    CALL FUNCTION 'SVRS_GET_VERSION_DIRECTORY_46'
      EXPORTING
        objname      = lv_obj_name
        objtype      = lv_object_type
      TABLES
        lversno_list = lt_dummy
        version_list = lt_versions
      EXCEPTIONS
        OTHERS       = 1.
    CHECK sy-subrc = 0.

    CALL FUNCTION 'F4IF_INT_TABLE_VALUE_REQUEST'
      EXPORTING
        retfield  = 'VERSNO'
        dynpprog  = sy-cprog
        dynpnr    = sy-dynnr
        value_org = 'S'
      TABLES
        value_tab  = lt_versions
        return_tab = lt_f4_list
      EXCEPTIONS
        OTHERS     = 1.
    CHECK sy-subrc = 0.

    READ TABLE lt_f4_list INTO DATA(ls_f4) INDEX 1.
    CHECK sy-subrc = 0.

    ASSIGN COMPONENT e_fieldname OF STRUCTURE <fs_row> TO <fs_field>.
    IF sy-subrc = 0.
      <fs_field> = ls_f4-fieldval.
    ENDIF.

    " The row above is updated, but the grid's on-screen buffer is
    " separate from the bound table until explicitly refreshed.
    ls_stable-row = abap_true.
    ls_stable-col = abap_true.
    go_alv_grid->refresh_table_display( is_stable = ls_stable ).

    er_event_data->m_event_handled = abap_true.
  ENDMETHOD.

  METHOD on_data_changed.
    " Clear the red highlight the moment a previously-flagged cell's
    " value changes - re-validation of whether it's now actually valid
    " still only happens on EXEC, this just removes the stale marker.
    FIELD-SYMBOLS <fs_row> TYPE ts_multi_obj.
    DATA ls_stable TYPE lvc_s_stbl.

    LOOP AT er_data_changed->mt_good_cells INTO DATA(ls_mod) WHERE fieldname = 'CMP_VERSNO'.
      READ TABLE gt_multi_obj ASSIGNING <fs_row> INDEX ls_mod-row_id.
      IF sy-subrc = 0.
        CLEAR <fs_row>-cellcolor.
      ENDIF.
    ENDLOOP.

    IF go_alv_grid IS BOUND.
      ls_stable-row = abap_true.
      ls_stable-col = abap_true.
      go_alv_grid->refresh_table_display( is_stable = ls_stable ).
    ENDIF.
  ENDMETHOD.
ENDCLASS.

PARAMETERS: p_prog TYPE sy-repid   MODIF ID prg,
            p_clas TYPE seoclsname MODIF ID cls,
            p_func TYPE rs38l_fnam MODIF ID fnc.

SELECT-OPTIONS: s_meth FOR gv_meth_dummy MODIF ID mth NO INTERVALS.

PARAMETERS: p_file TYPE string OBLIGATORY DEFAULT 'C:\temp\'.
SELECTION-SCREEN END OF BLOCK b2.

SELECTION-SCREEN BEGIN OF BLOCK b3 WITH FRAME TITLE text-003.
PARAMETERS: rb_bloc RADIOBUTTON GROUP grp2 DEFAULT 'X' USER-COMMAND ucomm,
            rb_diff RADIOBUTTON GROUP grp2,
            rb_full RADIOBUTTON GROUP grp2.
SELECTION-SCREEN END OF BLOCK b3.

*&---------------------------------------------------------------------*
*& Screen / F4 Events
*&---------------------------------------------------------------------*
AT SELECTION-SCREEN OUTPUT.
  PERFORM screen_modify.

AT SELECTION-SCREEN ON VALUE-REQUEST FOR p_file.
  PERFORM f4_output_folder CHANGING p_file.

AT SELECTION-SCREEN ON VALUE-REQUEST FOR s_meth-low.
  PERFORM f4_method_name.

*&---------------------------------------------------------------------*
*& Main
*&---------------------------------------------------------------------*
START-OF-SELECTION.
  PERFORM main.

*&---------------------------------------------------------------------*
*&      Form  MAIN
*&---------------------------------------------------------------------*
*       Orchestrates the whole extraction. Each step is a self
*       contained FORM so new extraction sources / output formats can
*       be plugged in without touching the others.
*       TR mode always routes through the multi-TR ALV review screen
*       (RUN_MULTI_TR_FLOW). A whole class (no specific method chosen)
*       routes through the same screen via RUN_CLASS_ALV_FLOW, listing
*       its section includes and methods for the user to pick from.
*       Program/Function modes, and Class mode with specific methods
*       already chosen, are unchanged.
*&---------------------------------------------------------------------*
FORM main.
  PERFORM determine_extraction_method.

  IF rb_tr = 'X'.
    PERFORM run_multi_tr_flow.
  ELSEIF rb_clas = 'X' AND s_meth IS INITIAL.
    PERFORM run_class_alv_flow.
  ELSE.
    PERFORM determine_output_path.

    PERFORM build_object_list.
    CHECK gv_error = abap_false.

    PERFORM add_class_methods.

    PERFORM process_all_objects.

    PERFORM download_output_file.
  ENDIF.
ENDFORM.

*&---------------------------------------------------------------------*
*&      Form  SCREEN_MODIFY
*&---------------------------------------------------------------------*
*       Shows/hides selection-screen fields depending on the chosen
*       extraction source (Transport / Program / Class / Function).
*&---------------------------------------------------------------------*
FORM screen_modify.
  LOOP AT SCREEN.
    IF screen-group1 = 'TR'  OR screen-group1 = 'PRG' OR
       screen-group1 = 'CLS' OR screen-group1 = 'FNC' OR
       screen-group1 = 'MTH'.
      screen-active = 0.
    ENDIF.

    IF rb_tr = 'X' AND screen-group1 = 'TR'.
      CLEAR: p_prog, p_clas, p_func.
      screen-active = 1.
    ELSEIF rb_prog = 'X' AND screen-group1 = 'PRG'.
      CLEAR: so_tr, p_clas, p_func.
      screen-active = 1.
    ELSEIF rb_clas = 'X' AND ( screen-group1 = 'CLS' OR screen-group1 = 'MTH' ).
      CLEAR: p_prog, so_tr, p_func.
      screen-active = 1.
    ELSEIF rb_func = 'X' AND screen-group1 = 'FNC'.
      CLEAR: p_prog, p_clas, so_tr.
      screen-active = 1.
    ENDIF.

    MODIFY SCREEN.
  ENDLOOP.
ENDFORM.

*&---------------------------------------------------------------------*
*&      Form  DETERMINE_EXTRACTION_METHOD
*&---------------------------------------------------------------------*
FORM determine_extraction_method.
  IF rb_diff IS NOT INITIAL.
    gv_extr_method = 'DIFFERENTIAL'.
  ELSEIF rb_bloc IS NOT INITIAL.
    gv_extr_method = 'BLOCKS'.
  ELSEIF rb_full IS NOT INITIAL.
    gv_extr_method = 'FULL'.
  ENDIF.
ENDFORM.

*&---------------------------------------------------------------------*
*&      Form  DETERMINE_OUTPUT_PATH
*&---------------------------------------------------------------------*
*       Program/Class/Function modes only - TR mode uses
*       DETERMINE_MULTI_OUTPUT_PATH instead.
*&---------------------------------------------------------------------*
FORM determine_output_path.
  IF p_prog IS NOT INITIAL.
    gv_filename = |CODE_EXTRACT_{ p_prog }_{ gv_extr_method }.TXT|.
  ELSEIF p_func IS NOT INITIAL.
    gv_filename = |CODE_EXTRACT_{ p_func }_{ gv_extr_method }.TXT|.
  ELSEIF p_clas IS NOT INITIAL.
    gv_filename = |CODE_EXTRACT_{ p_clas }_{ gv_extr_method }.TXT|.
  ELSE.
    gv_filename = |CODE_EXTRACT.TXT|.
  ENDIF.

  PERFORM build_full_path.
ENDFORM.

*&---------------------------------------------------------------------*
*&      Form  DETERMINE_MULTI_OUTPUT_PATH
*&---------------------------------------------------------------------*
*       TR mode: with exactly one TR entered, keeps the original
*       single-TR naming (CODE_EXTRACT_<TR>_<METHOD>.TXT) unchanged;
*       with more than one, uses CODE_EXTRACT_MULTI_TO<n>_<METHOD>.TXT,
*       where <n> is the number of TRs actually entered (resolved so_tr
*       matches).
*&---------------------------------------------------------------------*
FORM determine_multi_output_path.
  IF lines( gt_entered_trs ) = 1.
    gv_filename = |CODE_EXTRACT_{ gt_entered_trs[ 1 ] }_{ gv_extr_method }.TXT|.
  ELSE.
    gv_filename = |CODE_EXTRACT_MULTI_TO{ lines( gt_entered_trs ) }_{ gv_extr_method }.TXT|.
  ENDIF.
  PERFORM build_full_path.
ENDFORM.

*&---------------------------------------------------------------------*
*&      Form  BUILD_FULL_PATH
*&---------------------------------------------------------------------*
*       Combines P_FILE + GV_FILENAME into GV_FULL_PATH. Shared by both
*       output-path FORMs above.
*&---------------------------------------------------------------------*
FORM build_full_path.
  gv_full_path = p_file.
  IF substring( val = gv_full_path off = strlen( gv_full_path ) - 1 ) <> '\'.
    gv_full_path = gv_full_path && '\'.
  ENDIF.
  gv_full_path = gv_full_path && gv_filename.
ENDFORM.

*&---------------------------------------------------------------------*
*&      Form  BUILD_OBJECT_LIST
*&---------------------------------------------------------------------*
*       Dispatches to the source-specific list builder (Program/Class/
*       Function only - TR mode never reaches this, see MAIN). Sets
*       GV_ERROR if the input could not be resolved, which stops MAIN.
*&---------------------------------------------------------------------*
FORM build_object_list.
  CLEAR: gt_e071, gv_error.

  IF rb_prog = 'X'.
    PERFORM build_object_list_from_prog.
  ELSEIF rb_clas = 'X'.
    PERFORM build_object_list_from_class.
  ELSEIF rb_func = 'X'.
    PERFORM build_object_list_from_func.
  ENDIF.
ENDFORM.

*&---------------------------------------------------------------------*
*&      Form  BUILD_OBJECT_LIST_FROM_PROG
*&---------------------------------------------------------------------*
FORM build_object_list_from_prog.
  DATA ls_e071 TYPE e071.

  SELECT SINGLE name, subc
    FROM trdir
    INTO @DATA(ls_trdir)
    WHERE name = @p_prog.

  IF sy-subrc <> 0.
    MESSAGE 'Program does not exist or is inactive.' TYPE 'S' DISPLAY LIKE 'E'.
    gv_error = abap_true.
    RETURN.
  ENDIF.

  CASE ls_trdir-subc.
    WHEN 'I'.                            "Include
      ls_e071-pgmid  = 'LIMU'.
      ls_e071-object = 'REPS'.
    WHEN OTHERS.
      PERFORM add_program_includes USING p_prog.
      CLEAR ls_e071.
      ls_e071-pgmid  = 'R3TR'.
      ls_e071-object = 'PROG'.
  ENDCASE.

  ls_e071-obj_name = p_prog.
  APPEND ls_e071 TO gt_e071.
ENDFORM.

*&---------------------------------------------------------------------*
*&      Form  ADD_PROGRAM_INCLUDES
*&---------------------------------------------------------------------*
FORM add_program_includes USING iv_program TYPE sy-repid.
  DATA: lt_includes TYPE STANDARD TABLE OF sy-repid,
        ls_e071     TYPE e071.

  CALL FUNCTION 'RS_GET_ALL_INCLUDES'
    EXPORTING
      program      = iv_program
    TABLES
      includetab   = lt_includes
    EXCEPTIONS
      not_existent = 1
      no_program   = 2
      OTHERS       = 3.

  CHECK sy-subrc = 0.

  LOOP AT lt_includes ASSIGNING FIELD-SYMBOL(<fs_incl>).
    CLEAR ls_e071.
    ls_e071-pgmid    = 'LIMU'.
    ls_e071-object   = 'REPS'.
    ls_e071-obj_name = <fs_incl>.
    APPEND ls_e071 TO gt_e071.
  ENDLOOP.
ENDFORM.

*&---------------------------------------------------------------------*
*&      Form  BUILD_OBJECT_LIST_FROM_CLASS
*&---------------------------------------------------------------------*
FORM build_object_list_from_class.
  SELECT SINGLE clsname, clstype
    FROM seoclass
    INTO @DATA(ls_class)
    WHERE clsname = @p_clas.

  IF sy-subrc <> 0.
    MESSAGE 'Class does not exist.' TYPE 'S' DISPLAY LIKE 'E'.
    gv_error = abap_true.
    RETURN.
  ENDIF.

  IF s_meth IS INITIAL.
    APPEND VALUE #( pgmid = 'R3TR' object = 'CLAS' obj_name = p_clas ) TO gt_e071.
    APPEND VALUE #( pgmid = 'LIMU' object = 'CPUB' obj_name = p_clas ) TO gt_e071.
    APPEND VALUE #( pgmid = 'LIMU' object = 'CPRO' obj_name = p_clas ) TO gt_e071.
    APPEND VALUE #( pgmid = 'LIMU' object = 'CPRI' obj_name = p_clas ) TO gt_e071.

    APPEND VALUE #( sign = 'I' option = 'EQ' low = p_clas ) TO gr_class_name.
  ELSE.
    LOOP AT s_meth INTO DATA(ls_method).
      APPEND VALUE #( pgmid = 'LIMU' object = 'METH'
                       obj_name = |{ p_clas WIDTH = 30 }{ ls_method-low }| ) TO gt_e071.
    ENDLOOP.
  ENDIF.
ENDFORM.

*&---------------------------------------------------------------------*
*&      Form  BUILD_OBJECT_LIST_FROM_FUNC
*&---------------------------------------------------------------------*
FORM build_object_list_from_func.
  SELECT SINGLE funcname
    FROM tfdir
    INTO @DATA(lv_funcname)
    WHERE funcname = @p_func.

  IF sy-subrc <> 0.
    MESSAGE 'Function Module does not exist.' TYPE 'S' DISPLAY LIKE 'E'.
    gv_error = abap_true.
    RETURN.
  ENDIF.

  APPEND VALUE #( pgmid = 'LIMU' object = 'FUNC' obj_name = p_func ) TO gt_e071.
ENDFORM.

*&---------------------------------------------------------------------*
*&      Form  ADD_CLASS_METHODS
*&---------------------------------------------------------------------*
*       For every CLAS entry already in the object list (irrespective
*       of where it came from - Program or Class input), also extract
*       and list all of its methods.
*&---------------------------------------------------------------------*
FORM add_class_methods.
  LOOP AT gt_e071 ASSIGNING FIELD-SYMBOL(<fs_class>) WHERE object = 'CLAS'.
    APPEND VALUE #( sign = 'I' option = 'EQ' low = <fs_class>-obj_name ) TO gr_class_name.
  ENDLOOP.

  CHECK gr_class_name IS NOT INITIAL.

  SELECT clsname, cmpname
    INTO TABLE @DATA(lt_methods)
    FROM seocompo
   WHERE clsname IN @gr_class_name
     AND cmptype = '1'.

  CHECK sy-subrc = 0.

  LOOP AT lt_methods INTO DATA(ls_meth).
    APPEND VALUE #( object = 'METH'
                     obj_name = |{ ls_meth-clsname WIDTH = 30 }{ ls_meth-cmpname }| ) TO gt_e071.
  ENDLOOP.
ENDFORM.

*&---------------------------------------------------------------------*
*&      Form  PROCESS_ALL_OBJECTS
*&---------------------------------------------------------------------*
FORM process_all_objects.
  DATA: ls_e071        TYPE e071,
        lv_no_override TYPE vrsd-versno.

  LOOP AT gt_e071 INTO ls_e071.
    PERFORM process_single_object USING ls_e071 lv_no_override.
  ENDLOOP.
ENDFORM.

*&---------------------------------------------------------------------*
*&      Form  PROCESS_SINGLE_OBJECT
*&---------------------------------------------------------------------*
*       End-to-end handling of one object: dependency JSON, header,
*       version retrieval, delta computation and output formatting.
*       Any CHECK failure below simply skips this object, matching the
*       original report's CONTINUE behaviour.
*
*       IV_COMPARE_VERSNO is only used by the multi-TR ALV flow: when
*       non-initial, it overrides the normal auto-detected "previous
*       version" with the user's chosen baseline from the ALV.
*       PROCESS_ALL_OBJECTS (Program/Class/Function mode) always passes
*       an untouched (therefore initial) local variable for it, so their
*       behavior is completely unchanged. Not declared OPTIONAL - some
*       ABAP versions parse the OPTIONAL addition after "TYPE <field>"
*       as introducing a further formal parameter rather than a keyword,
*       so both call sites always pass it explicitly instead.
*&---------------------------------------------------------------------*
FORM process_single_object USING is_e071 TYPE e071
                                  iv_compare_versno TYPE vrsd-versno.
  DATA: lv_objname          TYPE vrsd-objname,
        lv_object_type      TYPE e071-object,
        lt_version_list     TYPE tt_vrsd,
        ls_latest_ver       TYPE vrsd,
        ls_prev_ver         TYPE vrsd,
        lt_code_new         TYPE tt_code_line,
        lt_code_old         TYPE tt_code_line,
        lt_trdir_new        TYPE tt_trdir,
        lt_trdir_old        TYPE tt_trdir,
        lt_delta            TYPE tt_delta,
        lv_only_one_version TYPE abap_bool,
        lv_skip             TYPE abap_bool.

  lv_objname     = is_e071-obj_name.
  lv_object_type = is_e071-object.

  PERFORM build_dependency_input USING is_e071 CHANGING lv_skip.
  CHECK lv_skip = abap_false.

  PERFORM fetch_dependency_json.
  PERFORM append_object_header USING is_e071.

  IF lv_object_type = 'PROG'.
    lv_object_type = 'REPS'.
  ENDIF.

  PERFORM get_version_list USING lv_objname lv_object_type
                            CHANGING lt_version_list lv_skip.
  CHECK lv_skip = abap_false.

  IF iv_compare_versno IS NOT INITIAL.
    " User picked an explicit baseline in the ALV - skip the normal
    " auto-detect (and its "first version" messaging, which wouldn't be
    " accurate here) but still drop TOC versions for a clean latest-version.
    PERFORM filter_toc_versions CHANGING lt_version_list.
    lv_only_one_version = abap_false.
  ELSE.
    PERFORM determine_comparison_mode CHANGING lt_version_list lv_only_one_version.
  ENDIF.

  READ TABLE lt_version_list INTO ls_latest_ver INDEX 1.
  CHECK sy-subrc = 0.

  PERFORM get_source_code USING lv_objname lv_object_type ls_latest_ver-versno
                           CHANGING lt_code_new lt_trdir_new lv_skip.
  CHECK lv_skip = abap_false.

  IF lv_only_one_version = abap_false.
    IF iv_compare_versno IS NOT INITIAL.
      ls_prev_ver-versno = iv_compare_versno.
    ELSE.
      READ TABLE lt_version_list INTO ls_prev_ver INDEX 2.
      CHECK sy-subrc = 0.
    ENDIF.

    PERFORM get_source_code USING lv_objname lv_object_type ls_prev_ver-versno
                             CHANGING lt_code_old lt_trdir_old lv_skip.
    CHECK lv_skip = abap_false.

    PERFORM compute_delta USING lt_code_old lt_code_new lt_trdir_old lt_trdir_new
                           CHANGING lt_delta lv_skip.
    CHECK lv_skip = abap_false.
  ELSE.
    PERFORM build_full_delta USING lt_code_new CHANGING lt_delta.
  ENDIF.

  IF rb_diff IS NOT INITIAL OR rb_full IS NOT INITIAL.
    PERFORM append_differential_output USING lt_delta.
  ENDIF.

  IF rb_bloc IS NOT INITIAL.
    PERFORM append_block_output USING lt_code_old lt_code_new lt_delta.
  ENDIF.

  APPEND VALUE #( line = | | ) TO gt_output.
ENDFORM.

*&---------------------------------------------------------------------*
*&      Form  BUILD_DEPENDENCY_INPUT
*&---------------------------------------------------------------------*
*       Resolves the concrete PROG/include that ZCR_GET_DEPENDENCY_OBJ_NEW
*       should analyse for the given transport object, into GT_DEP.
*       NOTE: the CPRI/CPRO mapping below intentionally mirrors the
*       original report's get_prosec_name/get_prisec_name pairing -
*       verify with the business if this was meant to be swapped.
*&---------------------------------------------------------------------*
FORM build_dependency_input USING is_e071 TYPE e071
                             CHANGING cv_skip TYPE abap_bool.
  DATA: lv_objname    TYPE vrsd-objname,
        lv_obj        TYPE string,
        lv_classname  TYPE string,
        lv_methodname TYPE string,
        lv_class_name TYPE seoclsname,
        ls_mdtkey     TYPE seocpdkey,
        lv_fm_name    TYPE rs38l_fnam,
        lv_fm_include TYPE progname.

  CLEAR: gt_dep, cv_skip.
  lv_objname = is_e071-obj_name.

  CASE is_e071-object.
    WHEN 'METH'.
      lv_obj = lv_objname.
      CONDENSE lv_obj.
      SPLIT lv_obj AT space INTO lv_classname lv_methodname.
      ls_mdtkey-clsname = lv_classname.
      ls_mdtkey-cpdname = lv_methodname.

      CALL METHOD cl_oo_classname_service=>get_method_include
        EXPORTING
          mtdkey              = ls_mdtkey
        RECEIVING
          result              = DATA(lv_include_name)
        EXCEPTIONS
          class_not_existing  = 1
          method_not_existing = 2
          OTHERS              = 3.
      IF sy-subrc <> 0.
        cv_skip = abap_true.
        RETURN.
      ENDIF.
      APPEND VALUE #( obj_type = 'PROG' object_name = lv_include_name ) TO gt_dep.

    WHEN 'FUNC'.
      lv_fm_name = lv_objname.
      CALL FUNCTION 'FUNCTION_INCLUDE_INFO'
        CHANGING
          funcname            = lv_fm_name
          include             = lv_fm_include
        EXCEPTIONS
          function_not_exists = 1
          include_not_exists  = 2
          group_not_exists    = 3
          no_selections       = 4
          no_function_include = 5
          OTHERS              = 6.
      IF sy-subrc <> 0.
        cv_skip = abap_true.
        RETURN.
      ENDIF.
      APPEND VALUE #( obj_type = 'PROG' object_name = lv_fm_include ) TO gt_dep.

    WHEN 'REPS'.
      APPEND VALUE #( obj_type = 'PROG' object_name = lv_objname ) TO gt_dep.

    WHEN 'CPUB'.
      lv_class_name = lv_objname.
      APPEND VALUE #( obj_type = 'PROG'
                       object_name = cl_oo_classname_service=>get_pubsec_name( lv_class_name ) ) TO gt_dep.

    WHEN 'CPRI'.
      lv_class_name = lv_objname.
      APPEND VALUE #( obj_type = 'PROG'
                       object_name = cl_oo_classname_service=>get_prosec_name( lv_class_name ) ) TO gt_dep.

    WHEN 'CPRO'.
      lv_class_name = lv_objname.
      APPEND VALUE #( obj_type = 'PROG'
                       object_name = cl_oo_classname_service=>get_prisec_name( lv_class_name ) ) TO gt_dep.

    WHEN 'CLSD'.
      lv_class_name = lv_objname.
      APPEND VALUE #( obj_type = 'PROG'
                       object_name = cl_oo_classname_service=>get_ccdef_name( lv_class_name ) ) TO gt_dep.

    WHEN OTHERS.
      APPEND VALUE #( obj_type = is_e071-object object_name = lv_objname ) TO gt_dep.
  ENDCASE.
ENDFORM.

*&---------------------------------------------------------------------*
*&      Form  FILTER_TOC_VERSIONS
*&---------------------------------------------------------------------*
*       Discards transport-of-copies (TOC) versions and versions with no
*       request number. Version order is: 000 -> active, 003 -> just
*       before, 002, 001. The comparison should happen between 000 and
*       003, but 000's request number can be empty (identical to 003) -
*       in that case we fall back to comparing 003 and 002, hence
*       versions from TOCs (or with an empty request) are dropped here.
*       Extracted out of DETERMINE_COMPARISON_MODE so the multi-TR ALV
*       flow (GET_DEFAULT_VERSIONS) and the version-override path in
*       PROCESS_SINGLE_OBJECT can reuse the same filtering without the
*       "This is the first version" messaging that only makes sense for
*       the auto-detect case.
*&---------------------------------------------------------------------*
FORM filter_toc_versions CHANGING ct_version_list TYPE tt_vrsd.
  DATA: lt_trkorr TYPE TABLE OF e070-trkorr,
        lt_e070   TYPE STANDARD TABLE OF ts_e070_min,
        ls_e070   TYPE ts_e070_min.

  lt_trkorr = VALUE #( FOR ls_ver IN ct_version_list ( ls_ver-korrnum ) ).
  DELETE ADJACENT DUPLICATES FROM lt_trkorr.

  SELECT trkorr, trfunction
    INTO TABLE @lt_e070
    FROM e070
    FOR ALL ENTRIES IN @lt_trkorr
   WHERE trkorr = @lt_trkorr-table_line.

  LOOP AT ct_version_list ASSIGNING FIELD-SYMBOL(<fs_version>).
    READ TABLE lt_e070 INTO ls_e070 WITH KEY trkorr = <fs_version>-korrnum.
    IF sy-subrc <> 0 OR ls_e070-trfunction = 'T'.
      CLEAR <fs_version>-korrnum.
    ENDIF.
  ENDLOOP.

  DELETE ct_version_list WHERE korrnum IS INITIAL.
ENDFORM.

*&---------------------------------------------------------------------*
*&      Form  DETERMINE_COMPARISON_MODE
*&---------------------------------------------------------------------*
*       Auto-detect path (Program/Class/Function mode, and TR-ALV mode
*       when no version override was chosen): filters TOC versions, then
*       decides whether this object only has one usable version to show
*       (in which case there is nothing to diff against).
*&---------------------------------------------------------------------*
FORM determine_comparison_mode CHANGING ct_version_list     TYPE tt_vrsd
                                         cv_only_one_version TYPE abap_bool.
  CLEAR cv_only_one_version.

  PERFORM filter_toc_versions CHANGING ct_version_list.

  IF lines( ct_version_list ) < 2.
    APPEND VALUE #( line = |This is the first version of this object.| ) TO gt_output.
    APPEND VALUE #( line = | | ) TO gt_output.
    cv_only_one_version = abap_true.
  ENDIF.

  IF rb_full IS NOT INITIAL.
    cv_only_one_version = abap_true.
  ENDIF.
ENDFORM.

*&---------------------------------------------------------------------*

*&---------------------------------------------------------------------*
*&      Form  BUILD_FULL_DELTA
*&---------------------------------------------------------------------*
*       Used for RB_FULL and for objects that only have one version:
*       treats every line of the current source as an insertion.
*&---------------------------------------------------------------------*
FORM build_full_delta USING it_code_new TYPE tt_code_line
                      CHANGING ct_delta TYPE tt_delta.
  DATA lv_tabix TYPE i.

  CLEAR ct_delta.

  LOOP AT it_code_new ASSIGNING FIELD-SYMBOL(<fs_code_new>).
    lv_tabix = sy-tabix.
    APPEND VALUE #( line = <fs_code_new>-line number = lv_tabix vrsflag = 'I' ) TO ct_delta.
  ENDLOOP.
ENDFORM.

*&---------------------------------------------------------------------*
*&      Form  APPEND_DIFFERENTIAL_OUTPUT
*&---------------------------------------------------------------------*
*       DIFFERENTIAL / FULL mode: one line per delta entry.
*&---------------------------------------------------------------------*
FORM append_differential_output USING it_delta TYPE tt_delta.
  DATA ls_delta TYPE ts_delta.

  LOOP AT it_delta INTO ls_delta.
    CASE ls_delta-vrsflag(1).
      WHEN 'I'.                        "Insertion
        APPEND VALUE #( line = |LINE { sy-tabix }: + { ls_delta-line }| ) TO gt_output.
      WHEN 'D'.                        "Deletion
        APPEND VALUE #( line = |LINE { sy-tabix }: - { ls_delta-line }| ) TO gt_output.
      WHEN 'M' OR 'U'.                 "Modified
        APPEND VALUE #( line = |LINE { sy-tabix }: ~ { ls_delta-line }| ) TO gt_output.
      WHEN OTHERS.
        CONTINUE.
    ENDCASE.
  ENDLOOP.
ENDFORM.

*&---------------------------------------------------------------------*
*&      Form  APPEND_BLOCK_OUTPUT
*&---------------------------------------------------------------------*
*       BLOCKS mode: rebuild a unified (old+new) line stream, then
*       print only the FORM/METHOD/FUNCTION/MODULE/DEFINE blocks that
*       actually contain a change.
*&---------------------------------------------------------------------*
FORM append_block_output USING it_code_old TYPE tt_code_line
                                it_code_new TYPE tt_code_line
                                it_delta    TYPE tt_delta.
  DATA lt_unified TYPE tt_unified.

  PERFORM build_unified_code USING it_code_old it_code_new it_delta
                              CHANGING lt_unified.
  PERFORM extract_changed_blocks USING lt_unified.
ENDFORM.

*&---------------------------------------------------------------------*
*&      Form  BUILD_UNIFIED_CODE
*&---------------------------------------------------------------------*
FORM build_unified_code USING it_code_old TYPE tt_code_line
                               it_code_new TYPE tt_code_line
                               it_delta    TYPE tt_delta
                        CHANGING ct_unified TYPE tt_unified.
  DATA: lv_idx_old  TYPE i VALUE 1,
        lv_idx_new  TYPE i VALUE 1,
        ls_code_old TYPE ts_code_line,
        ls_code_new TYPE ts_code_line,
        ls_delta    TYPE ts_delta.

  CLEAR ct_unified.

  LOOP AT it_delta INTO ls_delta.
    CASE ls_delta-vrsflag(1).
      WHEN 'D'.                        " Deletion
        WHILE lv_idx_old < ls_delta-number AND lv_idx_old <= lines( it_code_old ).
          READ TABLE it_code_old INTO ls_code_old INDEX lv_idx_old.
          APPEND VALUE #( flag = ' ' line = ls_code_old-line ) TO ct_unified.
          lv_idx_old = lv_idx_old + 1.
          lv_idx_new = lv_idx_new + 1.
        ENDWHILE.

        APPEND VALUE #( flag = 'D' line = ls_delta-line ) TO ct_unified.
        lv_idx_old = lv_idx_old + 1.

      WHEN 'I'.                        " Insertion
        WHILE lv_idx_new < ls_delta-number AND lv_idx_new <= lines( it_code_new ).
          READ TABLE it_code_new INTO ls_code_new INDEX lv_idx_new.
          APPEND VALUE #( flag = ' ' line = ls_code_new-line ) TO ct_unified.
          lv_idx_new = lv_idx_new + 1.
          lv_idx_old = lv_idx_old + 1.
        ENDWHILE.

        APPEND VALUE #( flag = 'I' line = ls_delta-line ) TO ct_unified.
        lv_idx_new = lv_idx_new + 1.

      WHEN 'M' OR 'U'.                 " Modified/Updated
        WHILE lv_idx_new < ls_delta-number AND lv_idx_new <= lines( it_code_new ).
          READ TABLE it_code_new INTO ls_code_new INDEX lv_idx_new.
          APPEND VALUE #( flag = ' ' line = ls_code_new-line ) TO ct_unified.
          lv_idx_new = lv_idx_new + 1.
          lv_idx_old = lv_idx_old + 1.
        ENDWHILE.

        APPEND VALUE #( flag = '~' line = ls_delta-line ) TO ct_unified.
        lv_idx_new = lv_idx_new + 1.
        lv_idx_old = lv_idx_old + 1.
    ENDCASE.
  ENDLOOP.

  " Catch any remaining unchanged lines at the end of the file
  WHILE lv_idx_new <= lines( it_code_new ).
    READ TABLE it_code_new INTO ls_code_new INDEX lv_idx_new.
    APPEND VALUE #( flag = ' ' line = ls_code_new-line ) TO ct_unified.
    lv_idx_new = lv_idx_new + 1.
  ENDWHILE.
ENDFORM.

*&---------------------------------------------------------------------*
*&      Form  EXTRACT_CHANGED_BLOCKS
*&---------------------------------------------------------------------*
*       Parses the unified line stream: tracks entry/exit of
*       FORM/METHOD/FUNCTION/MODULE/DEFINE blocks and only writes a
*       block to GT_OUTPUT if at least one line inside it changed.
*       Lines outside any block are written directly when changed.
*&---------------------------------------------------------------------*
FORM extract_changed_blocks USING it_unified TYPE tt_unified.
  DATA: lt_block_buffer  TYPE tt_output,
        lv_in_block      TYPE abap_bool,
        lv_block_changed TYPE abap_bool,
        lv_block_name    TYPE string,
        lv_text          TYPE string,
        lv_w1            TYPE string,
        lv_w2            TYPE string,
        lv_line_no       TYPE i,
        ls_unified       TYPE ts_unified,
        ls_output        TYPE ts_output.

  CLEAR: lt_block_buffer, lv_in_block, lv_block_changed.

  LOOP AT it_unified INTO ls_unified.

    " Capture the real line number FIRST - the block-header APPEND below
    " (into LT_BLOCK_BUFFER, a different table) resets SY-TABIX as a
    " side effect, which previously corrupted the "LINE n" label on the
    " very first line of every block (always showed as "LINE 1").
    lv_line_no = sy-tabix.

    lv_text = ls_unified-line.
    CONDENSE lv_text.

    " Extract the first keyword, ignoring full-line comments
    CLEAR lv_w1.
    IF lv_text IS NOT INITIAL AND lv_text(1) <> '*' AND lv_text(1) <> '"'.
      SPLIT lv_text AT space INTO lv_w1 lv_w2.
      TRANSLATE lv_w1 TO UPPER CASE.
      REPLACE ALL OCCURRENCES OF '.' IN lv_w1 WITH space.
      CONDENSE lv_w1.
    ENDIF.

    " Start of a measurable block?
    IF lv_in_block = abap_false AND
       ( lv_w1 = 'FORM' OR lv_w1 = 'METHOD' OR lv_w1 = 'FUNCTION' OR lv_w1 = 'MODULE' OR lv_w1 = 'DEFINE' ).
      lv_in_block = abap_true.
      CLEAR: lt_block_buffer, lv_block_changed.

      lv_block_name = lv_w2.
      REPLACE ALL OCCURRENCES OF '.' IN lv_block_name WITH space.
      CONDENSE lv_block_name.

      APPEND VALUE #( line = |--- Block: { lv_w1 } { lv_block_name } ---| ) TO lt_block_buffer.
    ENDIF.

    " Format the output line based on flag
    CLEAR ls_output.
    CASE ls_unified-flag.
      WHEN 'I'.
        ls_output-line = |LINE { lv_line_no }: + { ls_unified-line }|.
        lv_block_changed = abap_true.
      WHEN 'D'.
        ls_output-line = |LINE { lv_line_no }: - { ls_unified-line }|.
        lv_block_changed = abap_true.
      WHEN '~'.
        ls_output-line = |LINE { lv_line_no }: ~ { ls_unified-line }|.
        lv_block_changed = abap_true.
      WHEN OTHERS.
        ls_output-line = |LINE { lv_line_no }: = { ls_unified-line }|.
    ENDCASE.

    IF lv_in_block = abap_true.
      " Inside a block: accumulate all lines (changed and unchanged)
      APPEND ls_output TO lt_block_buffer.

      " End of the block?
      IF lv_w1 = 'ENDFORM' OR lv_w1 = 'ENDMETHOD' OR lv_w1 = 'ENDFUNCTION' OR
         lv_w1 = 'ENDMODULE' OR lv_w1 = 'END-OF-DEFINITION'.
        lv_in_block = abap_false.

        " Only push block to output if it was flagged as dirty
        IF lv_block_changed = abap_true.
          APPEND LINES OF lt_block_buffer TO gt_output.
          APPEND VALUE #( line = | | ) TO gt_output.
        ENDIF.
        CLEAR: lt_block_buffer, lv_block_changed.
      ENDIF.

    ELSE.
      " Outside a block (e.g. global data declarations)
      IF ls_unified-flag <> ' '.
        APPEND ls_output TO gt_output.
      ENDIF.
    ENDIF.

  ENDLOOP.

  " Flush buffer in case the block didn't properly terminate
  IF lv_in_block = abap_true AND lv_block_changed = abap_true.
    APPEND LINES OF lt_block_buffer TO gt_output.
    APPEND VALUE #( line = | | ) TO gt_output.
  ENDIF.
ENDFORM.

*&=======================================================================*
*& Multi-TR ALV review screen (TR mode only)
*&=======================================================================*

*&---------------------------------------------------------------------*
*&      Form  RUN_MULTI_TR_FLOW
*&---------------------------------------------------------------------*
*       Entry point for TR mode: build the union object list across all
*       entered TRs, show it on the ALV review screen (9000), then
*       process only the checked rows once the user presses Execute.
*       Back/Exit/Cancel on the screen abandon the run entirely - no
*       objects are processed and no file is written.
*&---------------------------------------------------------------------*
FORM run_multi_tr_flow.
  PERFORM validate_multi_tr_input.
  CHECK gv_error = abap_false.

  PERFORM resolve_entered_trs.
  CHECK gv_error = abap_false.

  PERFORM build_multi_tr_object_list.
  CHECK gv_error = abap_false.

  PERFORM determine_multi_output_path.

  CLEAR gv_multi_cancelled.
  CALL SCREEN 9000.

  CHECK gv_multi_cancelled = abap_false.
  PERFORM download_output_file.
ENDFORM.

*&---------------------------------------------------------------------*
*&      Form  RUN_CLASS_ALV_FLOW
*&---------------------------------------------------------------------*
*       Entry point for Class mode when no specific method was chosen
*       (whole class): builds the class's includes/methods, shows them
*       on the SAME ALV review screen (9000) used for TR mode, then
*       processes only the checked rows once the user presses Execute.
*       Back/Exit/Cancel abandon the run entirely, same as TR mode.
*       Class mode WITH specific methods chosen skips this screen
*       entirely (see MAIN) - there's nothing to pick from in that case.
*&---------------------------------------------------------------------*
FORM run_class_alv_flow.
  PERFORM build_class_object_list.
  CHECK gv_error = abap_false.

  PERFORM determine_output_path.

  CLEAR gv_multi_cancelled.
  CALL SCREEN 9000.

  CHECK gv_multi_cancelled = abap_false.
  PERFORM download_output_file.
ENDFORM.

*&---------------------------------------------------------------------*
*&      Form  BUILD_CLASS_OBJECT_LIST
*&---------------------------------------------------------------------*
*       Reuses BUILD_OBJECT_LIST (-> BUILD_OBJECT_LIST_FROM_CLASS) and
*       ADD_CLASS_METHODS exactly as the non-ALV Class flow does, then
*       converts the resulting GT_E071 rows into GT_MULTI_OBJ rows for
*       the ALV. The bare CLAS entry itself is skipped - it's the class
*       object, not one of its includes; only the definition/public/
*       protected/private section includes and the individual methods
*       are shown as checkable rows. ENRICH_MULTI_OBJECTS (shared with
*       TR mode) fills in each row's current/compare-against version.
*&---------------------------------------------------------------------*
FORM build_class_object_list.
  CLEAR gt_multi_obj.

  PERFORM build_object_list.
  CHECK gv_error = abap_false.

  PERFORM add_class_methods.

  LOOP AT gt_e071 INTO DATA(ls_e071) WHERE object <> 'CLAS'.
    APPEND VALUE #( checked     = abap_true
                     pgmid       = ls_e071-pgmid
                     object      = ls_e071-object
                     obj_name    = ls_e071-obj_name
                     obj_display = |{ ls_e071-obj_name } { ls_e071-object }| ) TO gt_multi_obj.
  ENDLOOP.

  IF gt_multi_obj IS INITIAL.
    MESSAGE 'No includes found for this class.' TYPE 'S' DISPLAY LIKE 'E'.
    gv_error = abap_true.
    RETURN.
  ENDIF.

  PERFORM enrich_multi_objects.
ENDFORM.

*&---------------------------------------------------------------------*
*&      Form  RESOLVE_ENTERED_TRS
*&---------------------------------------------------------------------*

*&---------------------------------------------------------------------*
*&      Form  BUILD_MULTI_TR_OBJECT_LIST
*&---------------------------------------------------------------------*
*       Builds GT_MULTI_OBJ: the union of objects across all entered
*       TRs, deduplicated by object (name + type), each enriched with
*       its TR list, most-recent TR, current version and default
*       compare-against version.
*&---------------------------------------------------------------------*
FORM build_multi_tr_object_list.
  DATA: lt_hits   TYPE tt_tr_hit,
        lv_trkorr TYPE e070-trkorr.

  CLEAR: gt_multi_obj, gv_error.

  LOOP AT gt_entered_trs INTO lv_trkorr.
    PERFORM collect_tr_hits USING lv_trkorr CHANGING lt_hits.
  ENDLOOP.

  IF lt_hits IS INITIAL.
    MESSAGE 'No Programs, Classes, or FMs found in the entered Transports.' TYPE 'S' DISPLAY LIKE 'E'.
    gv_error = abap_true.
    RETURN.
  ENDIF.

  PERFORM expand_class_methods CHANGING lt_hits.

  PERFORM build_unique_multi_objects USING lt_hits.
  PERFORM enrich_multi_objects.
ENDFORM.

*&---------------------------------------------------------------------*
*&      Form  EXPAND_CLASS_METHODS
*&---------------------------------------------------------------------*
*       Multi-TR equivalent of ADD_CLASS_METHODS: for every CLAS hit in
*       CT_HITS, also add its individual methods as METH hits - each
*       method inherits every TR its owning class hit appeared under, so
*       a class touched by two entered TRs gets its methods attributed
*       to both TRs too (matching the class's own TR-list behavior).
*       Reads a snapshot of the CLAS hits and builds new rows separately
*       before appending once, rather than appending into CT_HITS while
*       looping over it.
*&---------------------------------------------------------------------*
FORM expand_class_methods CHANGING ct_hits TYPE tt_tr_hit.
  DATA: lt_clas_hits  TYPE tt_tr_hit,
        lr_class_name TYPE RANGE OF seoclsname,
        lt_new_hits   TYPE tt_tr_hit.

  lt_clas_hits = VALUE #( FOR ls_hit IN ct_hits WHERE ( object = 'CLAS' ) ( ls_hit ) ).
  CHECK lt_clas_hits IS NOT INITIAL.

  lr_class_name = VALUE #( FOR ls_clas IN lt_clas_hits
                            ( sign = 'I' option = 'EQ' low = ls_clas-obj_name ) ).

  SELECT clsname, cmpname
    INTO TABLE @DATA(lt_methods)
    FROM seocompo
   WHERE clsname IN @lr_class_name
     AND cmptype = '1'.

  CHECK sy-subrc = 0.

  LOOP AT lt_methods INTO DATA(ls_meth).
    LOOP AT lt_clas_hits INTO DATA(ls_clas_hit) WHERE obj_name = ls_meth-clsname.
      APPEND VALUE #( pgmid    = 'LIMU'
                       object   = 'METH'
                       obj_name = |{ ls_meth-clsname WIDTH = 30 }{ ls_meth-cmpname }|
                       trkorr   = ls_clas_hit-trkorr ) TO lt_new_hits.
    ENDLOOP.
  ENDLOOP.

  APPEND LINES OF lt_new_hits TO ct_hits.
ENDFORM.

*&---------------------------------------------------------------------*
*&      Form  COLLECT_TR_HITS
*&---------------------------------------------------------------------*
*       Same object-list logic as the original single-TR flow (same
*       type filter, same subtask expansion, same dedup), just tagging
*       every resulting object with which entered TR it came from and
*       appending into the shared hits table instead of GT_E071.
*&---------------------------------------------------------------------*
FORM collect_tr_hits USING iv_trkorr TYPE e070-trkorr
                     CHANGING ct_hits TYPE tt_tr_hit.
  DATA: lt_subtask TYPE STANDARD TABLE OF e070-trkorr,
        lt_e071    TYPE tt_e071.

  SELECT trkorr FROM e070 INTO TABLE lt_subtask WHERE strkorr = iv_trkorr.

  DATA(lr_subtask) = VALUE rseloption(
    FOR <fs_subtask> IN lt_subtask
    ( sign = 'I' option = 'EQ' low = <fs_subtask> ) ).

  CHECK lr_subtask IS NOT INITIAL.

  SELECT * FROM e071 INTO TABLE lt_e071
    WHERE trkorr IN lr_subtask
      AND pgmid  IN ('R3TR','LIMU')
      AND object IN ('PROG', 'REPS', 'METH', 'CLAS', 'CLSD', 'CPUB', 'CPRI', 'CPRO', 'FUNC', 'DOMA', 'DTEL', 'TABL', 'TTYP')
    ORDER BY pgmid object obj_name.

  CHECK sy-subrc = 0.

  DELETE ADJACENT DUPLICATES FROM lt_e071 COMPARING pgmid object obj_name.

  LOOP AT lt_e071 ASSIGNING FIELD-SYMBOL(<fs_e071>).
    APPEND VALUE #( pgmid    = <fs_e071>-pgmid
                     object   = <fs_e071>-object
                     obj_name = <fs_e071>-obj_name
                     trkorr   = iv_trkorr ) TO ct_hits.
  ENDLOOP.
ENDFORM.

*&---------------------------------------------------------------------*
*&      Form  BUILD_UNIQUE_MULTI_OBJECTS
*&---------------------------------------------------------------------*
*       Deduplicates IT_HITS by (object, obj_name) into GT_MULTI_OBJ,
*       one row per unique object, checkbox defaulted to checked.
*&---------------------------------------------------------------------*
FORM build_unique_multi_objects USING it_hits TYPE tt_tr_hit.
  DATA: lt_unique TYPE tt_tr_hit,
        ls_multi  TYPE ts_multi_obj.

  lt_unique = it_hits.
  SORT lt_unique BY object obj_name.
  DELETE ADJACENT DUPLICATES FROM lt_unique COMPARING object obj_name.

  LOOP AT lt_unique INTO DATA(ls_unique).
    CLEAR ls_multi.
    ls_multi-checked     = abap_true.
    ls_multi-pgmid       = ls_unique-pgmid.
    ls_multi-object      = ls_unique-object.
    ls_multi-obj_name    = ls_unique-obj_name.
    ls_multi-obj_display = |{ ls_unique-obj_name } { ls_unique-object }|.

    PERFORM build_tr_list USING ls_unique-object ls_unique-obj_name it_hits
                           CHANGING ls_multi-tr_list ls_multi-most_recent_tr.

    APPEND ls_multi TO gt_multi_obj.
  ENDLOOP.
ENDFORM.

*&---------------------------------------------------------------------*
*&      Form  BUILD_TR_LIST
*&---------------------------------------------------------------------*
*       For one object, builds the comma-separated list of entered TRs
*       it appeared in, and determines the most recent one by
*       E070-AS4DATE/AS4TIME (last-changed/released timestamp).
*&---------------------------------------------------------------------*
FORM build_tr_list USING iv_object    TYPE e071-object
                          iv_obj_name TYPE e071-obj_name
                          it_hits     TYPE tt_tr_hit
                   CHANGING cv_tr_list       TYPE string
                            cv_most_recent   TYPE e070-trkorr.
  DATA: lt_matching  TYPE STANDARD TABLE OF e070-trkorr,
        lt_e070      TYPE STANDARD TABLE OF ts_e070_recency,
        ls_e070      TYPE ts_e070_recency,
        lv_best_date TYPE e070-as4date,
        lv_best_time TYPE e070-as4time.

  CLEAR: cv_tr_list, cv_most_recent, lv_best_date, lv_best_time.

  LOOP AT it_hits INTO DATA(ls_hit) WHERE object = iv_object AND obj_name = iv_obj_name.
    APPEND ls_hit-trkorr TO lt_matching.
    IF cv_tr_list IS INITIAL.
      cv_tr_list = ls_hit-trkorr.
    ELSE.
      cv_tr_list = |{ cv_tr_list }, { ls_hit-trkorr }|.
    ENDIF.
  ENDLOOP.

  CHECK lt_matching IS NOT INITIAL.

  SELECT trkorr, as4date, as4time
    INTO TABLE @lt_e070
    FROM e070
    FOR ALL ENTRIES IN @lt_matching
   WHERE trkorr = @lt_matching-table_line.

  LOOP AT lt_e070 INTO ls_e070.
    IF cv_most_recent IS INITIAL
    OR ls_e070-as4date > lv_best_date
    OR ( ls_e070-as4date = lv_best_date AND ls_e070-as4time > lv_best_time ).
      cv_most_recent = ls_e070-trkorr.
      lv_best_date   = ls_e070-as4date.
      lv_best_time   = ls_e070-as4time.
    ENDIF.
  ENDLOOP.
ENDFORM.

*&---------------------------------------------------------------------*
*&      Form  ENRICH_MULTI_OBJECTS
*&---------------------------------------------------------------------*
*       Fills in CUR_VERSNO/CUR_DISPLAY/CMP_VERSNO for every row using
*       GET_DEFAULT_VERSIONS (the silent, ALV-prefill equivalent of
*       DETERMINE_COMPARISON_MODE).
*&---------------------------------------------------------------------*
FORM enrich_multi_objects.
  DATA: lv_object_type TYPE e071-object,
        lv_obj_name    TYPE vrsd-objname,
        lv_skip        TYPE abap_bool.

  LOOP AT gt_multi_obj ASSIGNING FIELD-SYMBOL(<fs_multi>).
    lv_object_type = <fs_multi>-object.
    IF lv_object_type = 'PROG'.
      lv_object_type = 'REPS'.
    ENDIF.

    " E071-OBJ_NAME (CHAR120) and VRSD-OBJNAME (CHAR110, GET_DEFAULT_
    " VERSIONS' formal parameter type) are different lengths - route
    " through this temp variable, typed to match the formal parameter
    " exactly, rather than passing <fs_multi>-obj_name directly (which
    " PERFORM/FORM rejects as incompatible).
    lv_obj_name = <fs_multi>-obj_name.

    PERFORM get_default_versions USING lv_obj_name lv_object_type
                                  CHANGING <fs_multi>-cur_versno
                                           <fs_multi>-cmp_versno
                                           lv_skip.

    IF lv_skip = abap_true.
      <fs_multi>-cur_display = 'No version history found'.
    ELSEIF <fs_multi>-most_recent_tr IS NOT INITIAL.
      " TR mode: MOST_RECENT_TR is populated.
      <fs_multi>-cur_display = |Version { <fs_multi>-cur_versno } (TR: { <fs_multi>-most_recent_tr })|.
    ELSE.
      " Class mode: no TR context for these rows.
      <fs_multi>-cur_display = |Version { <fs_multi>-cur_versno }|.
    ENDIF.
  ENDLOOP.
ENDFORM.

*&---------------------------------------------------------------------*
*&      Form  GET_DEFAULT_VERSIONS
*&---------------------------------------------------------------------*
*       Silent equivalent of DETERMINE_COMPARISON_MODE for ALV pre-fill:
*       same TOC-filtering and same "latest = index 1 / previous = index
*       2" default logic, but without writing anything to GT_OUTPUT
*       (that only happens for objects actually processed later).
*       EV_CMP_VERSNO stays initial when there's nothing to compare
*       against (single-version object, or FULL mode) - PROCESS_
*       SINGLE_OBJECT already treats an initial override as "no override,
*       auto-detect", so this flows through correctly either way.
*&---------------------------------------------------------------------*
FORM get_default_versions USING iv_objname     TYPE vrsd-objname
                                 iv_object_type TYPE e071-object
                          CHANGING ev_cur_versno TYPE vrsd-versno
                                   ev_cmp_versno TYPE vrsd-versno
                                   cv_skip       TYPE abap_bool.
  DATA: lt_version_list TYPE tt_vrsd,
        ls_ver          TYPE vrsd.

  CLEAR: ev_cur_versno, ev_cmp_versno, cv_skip.

  PERFORM get_version_list USING iv_objname iv_object_type
                            CHANGING lt_version_list cv_skip.
  CHECK cv_skip = abap_false.

  PERFORM filter_toc_versions CHANGING lt_version_list.

  READ TABLE lt_version_list INTO ls_ver INDEX 1.
  IF sy-subrc <> 0.
    cv_skip = abap_true.
    RETURN.
  ENDIF.
  ev_cur_versno = ls_ver-versno.

  IF rb_full IS INITIAL AND lines( lt_version_list ) >= 2.
    READ TABLE lt_version_list INTO ls_ver INDEX 2.
    IF sy-subrc = 0.
      ev_cmp_versno = ls_ver-versno.
    ENDIF.
  ENDIF.
ENDFORM.

*&---------------------------------------------------------------------*
*&      Form  CREATE_MULTI_TR_ALV
*&---------------------------------------------------------------------*
*       Builds the ALV Grid on screen 9000's custom container (name
*       'ALV_CONTAINER' - must be created in SE51). Called once, from
*       MODULE STATUS_9000 OUTPUT.
*&---------------------------------------------------------------------*
FORM create_multi_tr_alv.
  DATA ls_layout TYPE lvc_s_layo.
  DATA lt_fieldcat TYPE lvc_t_fcat.

  CREATE OBJECT go_container
    EXPORTING
      container_name = 'ALV_CONTAINER'.

  CREATE OBJECT go_alv_grid
    EXPORTING
      i_parent = go_container.

  CREATE OBJECT go_alv_events.
  SET HANDLER go_alv_events->on_f4 FOR go_alv_grid.
  SET HANDLER go_alv_events->on_data_changed FOR go_alv_grid.

  PERFORM build_multi_tr_fieldcat CHANGING lt_fieldcat.

  ls_layout-cwidth_opt = abap_true.
  ls_layout-ctab_fname = 'CELLCOLOR'.

  go_alv_grid->set_table_for_first_display(
    EXPORTING
      is_layout       = ls_layout
    CHANGING
      it_outtab       = gt_multi_obj
      it_fieldcatalog = lt_fieldcat ).

  " F4AVAILABL on the fieldcat alone is not enough to make ONF4 fire -
  " the field also has to be explicitly registered here.
  go_alv_grid->register_f4_for_fields(
    it_f4 = VALUE lvc_t_f4( ( fieldname = 'CMP_VERSNO' register = abap_true ) ) ).

  " Needed so DATA_CHANGED actually fires when a cell's value changes
  " (used to clear a stale red highlight as soon as the user edits it).
  go_alv_grid->register_edit_event( cl_gui_alv_grid=>mc_evt_modified ).
ENDFORM.

*&---------------------------------------------------------------------*
*&      Form  BUILD_MULTI_TR_FIELDCAT
*&---------------------------------------------------------------------*
*       Only CHECKED and CMP_VERSNO are editable; CMP_VERSNO also gets a
*       custom F4 (handled by LCL_ALV_EVENT_HANDLER->ON_F4).
*&---------------------------------------------------------------------*
FORM build_multi_tr_fieldcat CHANGING ct_fieldcat TYPE lvc_t_fcat.
  CLEAR ct_fieldcat.

  APPEND VALUE #( fieldname = 'CHECKED'     coltext = 'Include'
                   checkbox = abap_true edit = abap_true outputlen = 8 ) TO ct_fieldcat.
  APPEND VALUE #( fieldname = 'OBJ_DISPLAY' coltext = 'Object'
                   outputlen = 40 ) TO ct_fieldcat.
  APPEND VALUE #( fieldname = 'TR_LIST'     coltext = 'TRs Appeared In'
                   outputlen = 40 ) TO ct_fieldcat.
  APPEND VALUE #( fieldname = 'CUR_DISPLAY' coltext = 'Current Version / Most Recent TR'
                   outputlen = 40 ) TO ct_fieldcat.
  APPEND VALUE #( fieldname = 'CMP_VERSNO'  coltext = 'Compare Against Version'
                   edit = abap_true f4availabl = abap_true outputlen = 20 ) TO ct_fieldcat.
ENDFORM.

*&---------------------------------------------------------------------*
*&      Form  PROCESS_CHECKED_MULTI_OBJECTS
*&---------------------------------------------------------------------*
*       Feeds every checked row into the existing per-object pipeline,
*       using its (possibly user-edited) CMP_VERSNO as the compare-
*       against override. Unchecked rows are never touched.
*       Only reached once VALIDATE_CHECKED_VERSIONS has already
*       confirmed every checked row's CMP_VERSNO is valid (called from
*       MODULE USER_COMMAND_9000 before this FORM), so no re-validation
*       happens here.
*&---------------------------------------------------------------------*
FORM process_checked_multi_objects.
  DATA ls_e071 TYPE e071.

  LOOP AT gt_multi_obj INTO DATA(ls_multi) WHERE checked = abap_true.
    CLEAR ls_e071.
    ls_e071-pgmid    = ls_multi-pgmid.
    ls_e071-object   = ls_multi-object.
    ls_e071-obj_name = ls_multi-obj_name.

    PERFORM process_single_object USING ls_e071 ls_multi-cmp_versno.
  ENDLOOP.
ENDFORM.

*&---------------------------------------------------------------------*
*&      Form  VALIDATE_CHECKED_VERSIONS
*&---------------------------------------------------------------------*
*       Screen-level gate, called from MODULE USER_COMMAND_9000 when
*       EXEC is pressed: validates every checked row's CMP_VERSNO
*       up front. If any are invalid, red-highlights that cell, shows
*       an error message, and sets GV_ERROR so the caller can CHECK it
*       and stay on screen 9000 for correction, instead of writing
*       anything to the output file.
*&---------------------------------------------------------------------*
FORM validate_checked_versions.
  DATA: lv_valid     TYPE abap_bool,
        lv_bad_count TYPE i,
        lv_bad_name  TYPE string,
        ls_stable    TYPE lvc_s_stbl.

  CLEAR gv_error.

  " Clear any highlighting left over from a previous failed attempt, so
  " a row that's since been corrected loses its red highlight too.
  LOOP AT gt_multi_obj ASSIGNING FIELD-SYMBOL(<fs_reset>).
    CLEAR <fs_reset>-cellcolor.
  ENDLOOP.

  LOOP AT gt_multi_obj ASSIGNING FIELD-SYMBOL(<fs_multi>) WHERE checked = abap_true.
    PERFORM validate_compare_version USING <fs_multi> CHANGING lv_valid.

    IF lv_valid = abap_false.
      gv_error = abap_true.
      lv_bad_count = lv_bad_count + 1.
      IF lv_bad_name IS INITIAL.
        lv_bad_name = <fs_multi>-obj_display.
      ENDIF.

      APPEND VALUE #( fname = 'CMP_VERSNO' color = VALUE #( col = 6 int = 1 inv = 0 ) )
             TO <fs_multi>-cellcolor.
    ENDIF.
  ENDLOOP.

  IF gv_error = abap_true.
    IF lv_bad_count = 1.
      MESSAGE |Invalid compare version for { lv_bad_name }. Please correct it.|
        TYPE 'S' DISPLAY LIKE 'E'.
    ELSE.
      MESSAGE |{ lv_bad_count } rows have an invalid compare version (e.g. { lv_bad_name }). Please correct them.|
        TYPE 'S' DISPLAY LIKE 'E'.
    ENDIF.
  ENDIF.

  IF go_alv_grid IS BOUND.
    ls_stable-row = abap_true.
    ls_stable-col = abap_true.
    go_alv_grid->refresh_table_display( is_stable = ls_stable ).
  ENDIF.
ENDFORM.

*&---------------------------------------------------------------------*
*&      Form  VALIDATE_COMPARE_VERSION
*&---------------------------------------------------------------------*
*       A blank CMP_VERSNO is always valid (means "no override, let
*       PROCESS_SINGLE_OBJECT auto-detect as usual"). A non-blank one
*       must actually exist in the object's TOC-filtered version list -
*       this catches a bogus value typed directly into the editable
*       cell, bypassing the F4 popup that would only ever offer real
*       version numbers.
*&---------------------------------------------------------------------*
FORM validate_compare_version USING is_multi TYPE ts_multi_obj
                              CHANGING cv_valid TYPE abap_bool.
  DATA: lv_object_type  TYPE e071-object,
        lv_obj_name     TYPE vrsd-objname,
        lt_version_list TYPE tt_vrsd,
        lv_skip         TYPE abap_bool.

  IF is_multi-cmp_versno IS INITIAL.
    cv_valid = abap_true.
    RETURN.
  ENDIF.

  cv_valid = abap_false.

  lv_object_type = is_multi-object.
  IF lv_object_type = 'PROG'.
    lv_object_type = 'REPS'.
  ENDIF.

  lv_obj_name = is_multi-obj_name.

  PERFORM get_version_list USING lv_obj_name lv_object_type
                            CHANGING lt_version_list lv_skip.
  CHECK lv_skip = abap_false.

  PERFORM filter_toc_versions CHANGING lt_version_list.

  READ TABLE lt_version_list WITH KEY versno = is_multi-cmp_versno
    TRANSPORTING NO FIELDS.
  IF sy-subrc = 0.
    cv_valid = abap_true.
  ENDIF.
ENDFORM.

*&---------------------------------------------------------------------*
*& Screen 9000 PBO/PAI
*&---------------------------------------------------------------------*
*       Screen 9000 itself (a Custom Control named 'ALV_CONTAINER', and
*       a GUI status 'STATUS_9000' with function codes EXEC/BACK/EXIT/
*       CANC) must be created manually in SE51/SE41 - see delivery notes.
*       Flow logic needed on screen 9000:
*         PBO: MODULE status_9000.
*         PAI: MODULE user_command_9000.
*&---------------------------------------------------------------------*
MODULE status_9000 OUTPUT.
  SET PF-STATUS 'STATUS_9000'.

  IF go_container IS NOT BOUND.
    PERFORM create_multi_tr_alv.
  ENDIF.
ENDMODULE.

MODULE user_command_9000 INPUT.
  DATA lv_ok_code TYPE sy-ucomm.

  lv_ok_code = sy-ucomm.
  CLEAR sy-ucomm.

  CASE lv_ok_code.
    WHEN 'EXEC'.
      IF go_alv_grid IS BOUND.
        go_alv_grid->check_changed_data( ).
      ENDIF.

      PERFORM validate_checked_versions.
      CHECK gv_error = abap_false.

      PERFORM process_checked_multi_objects.
      LEAVE TO SCREEN 0.

    WHEN 'BACK' OR 'EXIT' OR 'CANC'.
      " Abandon the run entirely - nothing gets processed or written.
      gv_multi_cancelled = abap_true.
      LEAVE TO SCREEN 0.
  ENDCASE.
ENDMODULE.
