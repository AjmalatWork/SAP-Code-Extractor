FUNCTION zcr_ai_code_review.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  IMPORTING
*"     REFERENCE(IT_OUTPUT) TYPE  STRING_TABLE
*"     REFERENCE(IV_REVIEW_PATH) TYPE  STRING
*"  EXPORTING
*"     REFERENCE(EV_MESSAGE) TYPE  STRING
*"----------------------------------------------------------------------
* Orchestrates the direct-API AI code review, per the Step 2 GEM
* instructions (GEM\Step 2 - Code Review GEM (1).md), against the rule
* knowledge base (GEM\*.csv, embedded below - see section 2). Called
* once, at the end of ZCR_AURA_CODE_EXTRACTOR_V2, with the extract
* file's lines already in memory (IT_OUTPUT) - no re-read from disk.
*
* IT_OUTPUT is the flat extract, one or more objects, each shaped:
*   =========================================
*   Object: <name> <type>[ - <note>]
*   Dependency JSON: {...}
*   DDIC JSON: {...}                              (optional)
*   LINE <n>: + <code>   / - <code>  / = <code>    (repeated)
* An object with only a note (nothing changed / extraction skipped) has
* no LINE rows and is skipped - nothing to review.
*
* Per object: code lines are split into 500-line chunks (Step 2 GEM
* section 6) and reviewed one chunk at a time, IN ORDER, because each
* chunk's prompt carries the findings already reported for that object
* so the model can de-dup (section 13) - chunks of the SAME object
* can't run in parallel. Chunks of DIFFERENT objects could, but this FM
* calls ZCR_AI_REVIEW_CHUNK synchronously: STARTING NEW TASK needs a
* FORM callback, and FORMs can't live inside a FUNCTION.
*
* The rule knowledge base (9 CSVs, ~35KB total) is small next to the
* code, so it's sent inline on every chunk call rather than through
* Gemini's context-cache API - simpler, and each call is stateless
* regardless. Dependency/DDIC JSON is only sent on an object's first
* chunk, since it doesn't change chunk to chunk and is unrelated to
* de-dup.
*
* Output: one TSV (tab-separated, not Excel) at IV_REVIEW_PATH, header
* row + one row per finding, columns per Step 2 GEM section 18. Of the
* 13 columns, 9 come from the model (Rule File Name, Rule Serial
* Number, Rule Name/Heading, Rule Tag, Object impacted, Line Number(s),
* Error/Violation Remarks, Recommended Fix, Review Chunk); the other 4
* (Remediation Status, Developer Comments, Target Completion Date,
* Reviewer Sign-Off) are left blank for manual entry later - they still
* get a column, just no value yet.

  "=============================
  " Local types
  "=============================
  TYPES: BEGIN OF ty_object,
           name      TYPE string,
           dep_json   TYPE string,
           ddic_json  TYPE string,
           has_code   TYPE abap_bool,
           code_lines TYPE string_table,
         END OF ty_object,
         tt_object TYPE STANDARD TABLE OF ty_object WITH EMPTY KEY.

  DATA: lt_objects       TYPE tt_object,
        ls_object        TYPE ty_object,
        lv_kb_text       TYPE string,
        lv_sys_instruct  TYPE string,
        lt_all_findings  TYPE zcr_tt_ai_finding,
        lt_obj_findings  TYPE zcr_tt_ai_finding,
        lt_tsv           TYPE string_table,
        lv_line          TYPE string,
        lv_objects_done  TYPE i,
        lv_chunks_done   TYPE i,
        lv_findings_done TYPE i,
        lv_errors        TYPE i.

  CONSTANTS: lc_sep       TYPE string VALUE '=========================================',
             lc_chunk_size TYPE i VALUE 500,
             lc_tab       TYPE c LENGTH 1 VALUE cl_abap_char_utilities=>horizontal_tab.

  CLEAR ev_message.

  "=====================================================================
  " 1. Parse IT_OUTPUT into one entry per object
  "=====================================================================
  LOOP AT it_output INTO lv_line.
    IF lv_line = lc_sep.
      IF ls_object-name IS NOT INITIAL AND ls_object-has_code = abap_true.
        APPEND ls_object TO lt_objects.
      ENDIF.
      CLEAR ls_object.

    ELSEIF lv_line CP 'Object:*'.
      ls_object-name = lv_line.
      SHIFT ls_object-name LEFT BY 8 PLACES. " strip 'Object: '

    ELSEIF lv_line CP 'Dependency JSON:*'.
      ls_object-dep_json = lv_line.
      SHIFT ls_object-dep_json LEFT BY 17 PLACES.

    ELSEIF lv_line CP 'DDIC JSON:*'.
      ls_object-ddic_json = lv_line.
      SHIFT ls_object-ddic_json LEFT BY 11 PLACES.

    ELSEIF lv_line CP 'LINE *'.
      APPEND lv_line TO ls_object-code_lines.
      ls_object-has_code = abap_true.
    ENDIF.
  ENDLOOP.

  " Flush the last object - it never hits another separator line.
  IF ls_object-name IS NOT INITIAL AND ls_object-has_code = abap_true.
    APPEND ls_object TO lt_objects.
  ENDIF.

  IF lt_objects IS INITIAL.
    ev_message = 'AI review skipped - no reviewable code found in the extract.'.
    RETURN.
  ENDIF.

          " not the model - overwrite whatever came back.

    REPLACE ALL OCCURRENCES OF lc_tab IN ls_finding-object_name  WITH ' '.
    REPLACE ALL OCCURRENCES OF lc_tab IN ls_finding-line_nos     WITH ' '.
    REPLACE ALL OCCURRENCES OF lc_tab IN ls_finding-remarks      WITH ' '.
    REPLACE ALL OCCURRENCES OF lc_tab IN ls_finding-fix          WITH ' '.
    REPLACE ALL OCCURRENCES OF cl_abap_char_utilities=>newline IN ls_finding-remarks WITH ' '.
    REPLACE ALL OCCURRENCES OF cl_abap_char_utilities=>newline IN ls_finding-fix     WITH ' '.

    APPEND |{ ls_finding-rule_file }{ lc_tab }{ ls_finding-rule_serial }{ lc_tab }| &&
           |{ ls_finding-rule_heading }{ lc_tab }{ ls_finding-rule_tag }{ lc_tab }| &&
           |{ ls_finding-object_name }{ lc_tab }{ ls_finding-line_nos }{ lc_tab }| &&
           |{ ls_finding-remarks }{ lc_tab }{ ls_finding-fix }{ lc_tab }{ lc_tab }| &&
           |{ lc_tab }{ lc_tab }{ lc_tab }{ ls_finding-review_chunk }| TO lt_tsv.
  ENDLOOP.

  CALL FUNCTION 'GUI_DOWNLOAD'
    EXPORTING
      filename         = iv_review_path
      filetype         = 'ASC'
    TABLES
      data_tab         = lt_tsv
    EXCEPTIONS
      file_write_error = 1
      OTHERS           = 2.

  IF sy-subrc = 0.
    DATA(lv_error_note) = COND string( WHEN lv_errors > 0 THEN |, { lv_errors } chunk(s) failed| ELSE '' ).
    ev_message = |AI review done - { lv_kb_files_loaded } rule file(s) loaded, { lv_objects_done } object(s), | &&
                 |{ lv_chunks_done } chunk(s), { lv_findings_done } finding(s){ lv_error_note }. | &&
                 |Written to { iv_review_path }|.
  ELSE.
    ev_message = |AI review finished but the TSV could not be written to { iv_review_path }|.
  ENDIF.

ENDFUNCTION.
