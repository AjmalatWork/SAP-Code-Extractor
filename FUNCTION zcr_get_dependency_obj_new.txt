FUNCTION zcr_get_dependency_obj_new.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  IMPORTING
*"     REFERENCE(II_OBJECTS) TYPE  ZPOC_TT_OBJ
*"  EXPORTING
*"     REFERENCE(EV_JSON) TYPE  STRING
*"     REFERENCE(EV_DDIC_JSON) TYPE  STRING
*"----------------------------------------------------------------------
* NOTE: subroutines (FORM) cannot be nested inside FUNCTION/ENDFUNCTION,
* so this FM stays a single block. Repetition is instead avoided by having
* each object-type branch below only COLLECT raw dependency candidates
* into LT_CANDIDATES; the duplicate-check + signature enrichment + append
* logic that used to be repeated per branch now runs exactly once, in the
* shared loop after the CASE.

  "=============================
  " Type Definitions
  "=============================
  TYPES: BEGIN OF ty_dependency,
           type      TYPE string,
           name      TYPE string,
           signature TYPE zcr_ts_signature,
         END OF ty_dependency.

  TYPES: ty_t_dependency TYPE STANDARD TABLE OF ty_dependency WITH EMPTY KEY.

  TYPES: BEGIN OF ty_object_output,
           type         TYPE string,
           name         TYPE string,
           dependencies TYPE ty_t_dependency,
         END OF ty_object_output.

  TYPES: ty_t_output TYPE STANDARD TABLE OF ty_object_output WITH EMPTY KEY.

  " A raw dependency found by one of the branches below, before dedup/
  " enrichment. LOOKUP_NAME is what gets passed to ZCR_GET_OBJECT_SIGNATURE
  " (for interface methods this is the plain method name, while NAME keeps
  " the "INTERFACE~METHOD" form expected in the output). PARENT is only
  " set for METH candidates (owning class/interface, needed by RTTI).
  TYPES: BEGIN OF ty_candidate,
           type        TYPE string,
           name        TYPE string,
           lookup_name TYPE string,
           parent      TYPE string,
         END OF ty_candidate.

  TYPES: ty_t_candidate TYPE STANDARD TABLE OF ty_candidate WITH EMPTY KEY.

  "=============================
  " Data Declarations
  "=============================
  DATA: lt_output        TYPE ty_t_output,
        ls_output        TYPE ty_object_output,
        ls_dep           TYPE ty_dependency,
        lt_candidates    TYPE ty_t_candidate,
        ls_ddic_obj_list TYPE zpoc_ts_ddic_obj,
        li_ddic_obj_list TYPE zpoc_tt_ddic_obj.

  DATA: lt_env TYPE TABLE OF senvi,
        ls_env TYPE senvi.

  DATA: BEGIN OF ls_func,
          funcname TYPE tfdir-funcname,
        END OF ls_func,
        lt_funcs LIKE TABLE OF ls_func.

  DATA: BEGIN OF ls_include,
          include TYPE d010inc-include,
        END OF ls_include,
        lt_includes LIKE TABLE OF ls_include.

  DATA: lv_area TYPE progname.

  DATA: BEGIN OF ls_method,
          cmpname TYPE seocompo-cmpname,
          cmptype TYPE seocompo-cmptype,
        END OF ls_method,
        lt_methods LIKE TABLE OF ls_method.

  DATA: BEGIN OF ls_interface,
          refclsname TYPE seometarel-refclsname,
        END OF ls_interface,
        lt_interfaces LIKE TABLE OF ls_interface.

  DATA: BEGIN OF ls_intf_method,
          cmpname TYPE seocompo-cmpname,
          cmptype TYPE seocompo-cmptype,
        END OF ls_intf_method,
        lt_intf_methods LIKE TABLE OF ls_intf_method.

  "=============================
  " Main Processing Loop
  "=============================
  LOOP AT ii_objects INTO DATA(ls_input).

    CLEAR: ls_output, lt_env, lt_candidates.

    ls_output-type = ls_input-obj_type.
    ls_output-name = ls_input-object_name.

    CASE ls_input-obj_type.
      WHEN 'PROG'.

        CALL FUNCTION 'REPOSITORY_ENVIRONMENT_RFC'
          EXPORTING
            obj_type        = ls_input-obj_type
*           ENVIRONMENT_TYPES       =
            object_name     = ls_input-object_name
          TABLES
            environment_tab = lt_env
*           SOURCE_OBJECTS  =
          .

        IF sy-subrc <> 0.
          CONTINUE.
        ENDIF.

        LOOP AT lt_env INTO ls_env.

          " Skip empty entries
          IF ls_env-object IS INITIAL.
            CONTINUE.
          ENDIF.

          " Filter useful object types only
*        CASE ls_env-type.
*          WHEN 'PROG' OR 'REPS' OR 'FUGR' OR 'FUNC' OR 'TABL' OR 'CLAS' OR 'INTF'.

          " Skip non-custom objects
*          IF ls_env-object NP 'Z*' AND ls_env-object NP 'Y*'.
*            CONTINUE.
*          ENDIF.
*
*          " Keep only required types
*          IF ls_env-type <> 'INCL' AND ls_env-type <> 'FUNC' AND ls_env-type <> 'CLAS'.
*            CONTINUE.
*          ENDIF.

          " NOTE: METH candidates discovered here (if any) get no PARENT,
          " so the shared enrichment step below leaves them unenriched -
          " REPOSITORY_ENVIRONMENT_RFC doesn't reliably expose an owning
          " class name in a form RTTI can consume.
          APPEND VALUE #( type = ls_env-type name = ls_env-object
                           lookup_name = ls_env-object ) TO lt_candidates.

          "*****************************************
          " prepare List of  DDIC Objects
          "*****************************************
          IF ls_env-type = 'DOMA'  OR ls_env-type = 'DTEL' OR
             ls_env-type = 'TABL'  OR ls_env-type = 'TTYP'.
            CLEAR: ls_ddic_obj_list.
            ls_ddic_obj_list-obj_type = ls_env-type.
            ls_ddic_obj_list-object_name = ls_env-object.
            APPEND ls_ddic_obj_list TO li_ddic_obj_list.
          ENDIF.

        ENDLOOP.

      WHEN 'FUGR'.

        CONCATENATE 'SAPL' ls_input-object_name INTO lv_area.

*****************************************
* Get Function Modules of Function Group
*****************************************
        SELECT funcname
          INTO TABLE lt_funcs
          FROM tfdir
          WHERE pname = lv_area.

        LOOP AT lt_funcs INTO ls_func.

          IF ls_func-funcname NP 'Z*'
          AND ls_func-funcname NP 'Y*'.
            CONTINUE.
          ENDIF.

          APPEND VALUE #( type = 'FUNC' name = ls_func-funcname
                           lookup_name = ls_func-funcname ) TO lt_candidates.

        ENDLOOP.


*****************************************
* Get Includes of Function Group
*****************************************
        SELECT include
          INTO TABLE lt_includes
          FROM d010inc
          WHERE master = lv_area.

        LOOP AT lt_includes INTO ls_include.

          IF ls_include-include NP '*TOP'
          AND ls_include-include NP '*UXX'.
            CONTINUE.
          ENDIF.

          APPEND VALUE #( type = 'INCL' name = ls_include-include
                           lookup_name = ls_include-include ) TO lt_candidates.

        ENDLOOP.

      WHEN 'CLAS'.

        "*****************************************
        " Get Class Methods
        "*****************************************
        SELECT cmpname cmptype
          INTO TABLE lt_methods
          FROM seocompo
          WHERE clsname = ls_input-object_name
            AND cmptype = '1'.

        LOOP AT lt_methods INTO ls_method.

          APPEND VALUE #( type = 'METH' name = ls_method-cmpname
                           lookup_name = ls_method-cmpname
                           parent = ls_input-object_name ) TO lt_candidates.

        ENDLOOP.


        "*****************************************
        " Get Implemented Interfaces
        "*****************************************
        SELECT refclsname
          INTO TABLE lt_interfaces
          FROM seometarel
          WHERE clsname = ls_input-object_name
            AND reltype = '1'.

        LOOP AT lt_interfaces INTO ls_interface.

          " INTF as a referenced object is not enriched (only its member
          " methods are, when the interface itself is later processed).
          APPEND VALUE #( type = 'INTF' name = ls_interface-refclsname
                           lookup_name = ls_interface-refclsname ) TO lt_candidates.

        ENDLOOP.


      WHEN 'INTF'.

        "*****************************************
        " Get Interface Methods
        "*****************************************
        SELECT cmpname cmptype
          INTO TABLE lt_intf_methods
          FROM seocompo
          WHERE clsname = ls_input-object_name
            AND cmptype = '1'.

        LOOP AT lt_intf_methods INTO ls_intf_method.

          APPEND VALUE #( type = 'METH'
                           name = |{ ls_input-object_name }~{ ls_intf_method-cmpname }|
                           lookup_name = ls_intf_method-cmpname
                           parent = ls_input-object_name ) TO lt_candidates.

        ENDLOOP.

      WHEN 'DOMA' OR  'DTEL' OR 'TABL' OR 'TTYP'.
        "*****************************************
        " prepare List of  DDIC Objects
        "*****************************************
        CLEAR: ls_ddic_obj_list.
        ls_ddic_obj_list-obj_type = ls_input-obj_type.
        ls_ddic_obj_list-object_name = ls_input-object_name.
        APPEND ls_ddic_obj_list TO li_ddic_obj_list.
        CONTINUE.
    ENDCASE.

    "=============================
    " Dedup + enrich + append (shared by every branch above)
    "=============================
    LOOP AT lt_candidates INTO DATA(ls_candidate).

      CLEAR ls_dep.
      ls_dep-type = ls_candidate-type.
      ls_dep-name = ls_candidate-name.

      " Avoid duplicates
      READ TABLE ls_output-dependencies
        WITH KEY type = ls_dep-type
                 name = ls_dep-name
        TRANSPORTING NO FIELDS.

      IF sy-subrc = 0.
        CONTINUE.
      ENDIF.

      " Enrich PROG/INCL/FUNC/METH with a signature summary. (DOMA/DTEL/
      " TABL/TTYP/INTF are intentionally excluded - DDIC objects get their
      " own richer metadata from Z_GET_DDIC_INFO instead, and INTF as a
      " referenced object has no signature of its own.)
      IF ls_candidate-type = 'PROG' OR ls_candidate-type = 'INCL' OR
         ls_candidate-type = 'FUNC' OR ls_candidate-type = 'METH'.
        CALL FUNCTION 'ZCR_GET_OBJECT_SIGNATURE'
          EXPORTING
            iv_obj_type    = ls_candidate-type
            iv_object_name = ls_candidate-lookup_name
            iv_parent_name = ls_candidate-parent
          IMPORTING
            es_signature   = ls_dep-signature.
      ENDIF.

      APPEND ls_dep TO ls_output-dependencies.

    ENDLOOP.

    APPEND ls_output TO lt_output.

  ENDLOOP.

  "=============================
  " Convert to JSON
  "=============================
  ev_json = /ui2/cl_json=>serialize(
    data     = lt_output
    compress = abap_true ).

  IF li_ddic_obj_list IS NOT INITIAL.
** Here We are retrieving all the DDIC Objects in JSON format.
** Returned as its own independent JSON in EV_DDIC_JSON - no longer
** string-concatenated onto EV_JSON.
    CALL FUNCTION 'Z_GET_DDIC_INFO'
      EXPORTING
        ii_objects = li_ddic_obj_list
      IMPORTING
        ew_json    = ev_ddic_json.
  ENDIF.

ENDFUNCTION.
