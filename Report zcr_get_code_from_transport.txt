" Structure for the reconstructed unified file

         flag TYPE c LENGTH 1,
         line TYPE ts_line_255,
       END OF ts_unified.

  ls_unified          TYPE ts_unified,

  lw_fm_name          TYPE rs38l_fnam,
  lw_fm_include       TYPE progname,
  lw_class_name       TYPE seoclsname.

  IF rb_tr = 'X'.
    " Check if TR was entered
    IF p_tr IS INITIAL.
      MESSAGE 'Please enter a Transport Request.' TYPE 'S' DISPLAY LIKE 'E'.
      RETURN.
    ENDIF.

    SELECT trkorr FROM e070 INTO TABLE @DATA(li_subtask)
      WHERE strkorr = @p_tr
      ORDER BY PRIMARY KEY.
    IF sy-subrc IS NOT INITIAL.
      MESSAGE 'Enter valid transport request' TYPE 'S' DISPLAY LIKE 'E'.
      RETURN.
    ENDIF.

    DATA: lr_subtask TYPE RANGE OF trkorr.

    LOOP AT li_subtask ASSIGNING FIELD-SYMBOL(<fs_subtask>).
      APPEND VALUE #( sign = 'I'
                      option = 'EQ'
                      low = <fs_subtask> ) TO lr_subtask.
    ENDLOOP.

    " Fetch all objects in the subtask
    SELECT * FROM e071 INTO TABLE li_e071
      WHERE trkorr IN lr_subtask
        AND pgmid  IN ('R3TR','LIMU')
        AND object IN ('PROG', 'REPS', 'METH', 'CLAS', 'CLSD', 'CPUB', 'CPRI', 'CPRO', 'FUNC', 'DOMA' , 'DTEL' , 'TABL' , 'TTYP')
      ORDER BY pgmid object obj_name.

    IF sy-subrc <> 0.
      MESSAGE 'No Programs, Classes, or FMs found in this Transport.' TYPE 'S' DISPLAY LIKE 'E'.
      RETURN.
    ENDIF.

    DELETE ADJACENT DUPLICATES FROM li_e071 COMPARING pgmid object obj_name.
    CLEAR: lr_subtask, li_subtask.

  ELSEIF rb_prog = 'X'.
    " Verify Program existence in TRDIR
    SELECT SINGLE name, subc
      FROM trdir
      INTO @DATA(ls_trdir)
      WHERE name = @p_prog.

    IF sy-subrc <> 0.
      MESSAGE 'Program does not exist or is inactive.' TYPE 'S' DISPLAY LIKE 'E'.
      RETURN.
    ENDIF.
    " Mock the E071 entry so downstream logic works
    CASE: ls_trdir-subc.
      WHEN 'I'.     "Include
        ls_e071-pgmid    = 'LIMU'.
        ls_e071-object   = 'REPS'.
      WHEN OTHERS.
        DATA: lv_program  TYPE sy-repid,
              lt_includes TYPE STANDARD TABLE OF sy-repid,
              ls_include  TYPE sy-repid.
        move p_prog to lv_program.
        CALL FUNCTION 'RS_GET_ALL_INCLUDES'
          EXPORTING
            program      = lv_program
          TABLES
            includetab   = lt_includes
          EXCEPTIONS
            not_existent = 1
            no_program   = 2
            OTHERS       = 3.

        IF sy-subrc = 0.
          LOOP AT lt_includes ASSIGNING FIELD-SYMBOL(<fs_incl>).
            ls_e071-pgmid    = 'LIMU'.
            ls_e071-object   = 'REPS'.
            ls_e071-obj_name = <fs_incl>.
            APPEND ls_e071 TO li_e071.
          ENDLOOP.
        ENDIF.

        CLEAR ls_e071.
        ls_e071-pgmid    = 'R3TR'.
        ls_e071-object   = 'PROG'.

    ENDCASE.
    ls_e071-obj_name = p_prog.
    APPEND ls_e071 TO li_e071.

  ELSEIF rb_clas = 'X'.
    " Verify Class existence in SEOCLASS
    SELECT SINGLE clsname, clstype
      FROM seoclass
      INTO @DATA(ls_class)
      WHERE clsname = @p_clas.
    IF sy-subrc <> 0.
      MESSAGE 'Class does not exist.' TYPE 'S' DISPLAY LIKE 'E'.
      RETURN.
    ENDIF.
    " Mock the E071 entry
    IF s_meth IS INITIAL.
      " Class
      APPEND VALUE #( pgmid = 'R3TR' object = 'CLAS' obj_name = p_clas ) TO li_e071.
      " Class Public Section
      APPEND VALUE #( pgmid = 'LIMU' object = 'CPUB' obj_name = p_clas ) TO li_e071.
      " Class Protected Section
      APPEND VALUE #( pgmid = 'LIMU' object = 'CPRO' obj_name = p_clas ) TO li_e071.
      " Class Private Section
      APPEND VALUE #( pgmid = 'LIMU' object = 'CPRI' obj_name = p_clas ) TO li_e071.

      APPEND INITIAL LINE TO lr_class_name ASSIGNING FIELD-SYMBOL(<fs_line>).
      <fs_line>-sign = 'I'.
      <fs_line>-option = 'EQ'.
      <fs_line>-low = p_clas.
    ELSE.
      " Method specific handling
      LOOP AT s_meth INTO DATA(ls_method).
        APPEND VALUE #( pgmid = 'LIMU' object = 'METH' obj_name = |{ p_clas WIDTH = 30 }{ ls_method-low }| ) TO li_e071.
      ENDLOOP.
    ENDIF.



  ELSEIF rb_func = 'X'.
    " Verify FM existence in TFDIR
    SELECT SINGLE funcname
      FROM tfdir
      INTO @DATA(lw_funcname)
      WHERE funcname = @p_func.

    IF sy-subrc <> 0.
      MESSAGE 'Function Module does not exist.' TYPE 'S' DISPLAY LIKE 'E'.
      RETURN.
    ENDIF.

    " Mock the E071 entry (FMs are technically LIMU FUNC in transports)
    ls_e071-pgmid    = 'LIMU'.
    ls_e071-object   = 'FUNC'.
    ls_e071-obj_name = p_func.
    APPEND ls_e071 TO li_e071.
  ENDIF.

*Here We are fetching List of Methods as well If the Class is having only the current Version
*As we need methods in this case
* For each object
  LOOP AT li_e071 ASSIGNING FIELD-SYMBOL(<fs_class>) WHERE object = 'CLAS'.

* Removing as multiple subtasks can have same new class and then this check fails
* Adding all methods of class instead for now

*    " New class -> only active version exists
*    IF p_tr IS NOT INITIAL.

** If Entry Found that means This is the existing Class

    APPEND VALUE #( sign   = 'I'
                    option = 'EQ'
                    low    = <fs_class>-obj_name ) TO lr_class_name.
  ENDLOOP.

  IF lr_class_name IS NOT INITIAL.
    SELECT clsname,cmpname
    INTO TABLE @DATA(li_methods)
    FROM seocompo
   WHERE clsname IN @lr_class_name
     AND cmptype = '1'.
    IF sy-subrc IS NOT INITIAL.
      RETURN.
    ENDIF.

    LOOP AT li_methods INTO DATA(ls_meth).
      CLEAR ls_e071.

      ls_e071-obj_name = |{ ls_meth-clsname WIDTH = 30 }{ ls_meth-cmpname }|.
      ls_e071-object   = 'METH'.

      APPEND ls_e071 TO li_e071.
    ENDLOOP.
  ENDIF.


  DATA: ls_mdtkey TYPE seocpdkey.

  LOOP AT li_e071 INTO ls_e071.

    MOVE ls_e071-obj_name TO lw_objname.

    CLEAR: li_versions, li_version_list, ls_latest_ver, ls_prev_ver,
           li_code_new, li_code_old, li_trdir_new, li_trdir_old,
           li_trdir_delta, li_delta, ls_output, lw_only_one_version, ls_mdtkey.

    IF ls_e071-object = 'METH'.
      DATA(lw_obj) = lw_objname.
      CONDENSE lw_obj.
      SPLIT lw_obj AT space INTO DATA(lw_classname) DATA(lw_methodname).
      ls_mdtkey-clsname = lw_classname.
      ls_mdtkey-cpdname = lw_methodname.
      CALL METHOD cl_oo_classname_service=>get_method_include
        EXPORTING
          mtdkey              = ls_mdtkey
        RECEIVING
          result              = DATA(lw_include_name)
        EXCEPTIONS
          class_not_existing  = 1
          method_not_existing = 2
          OTHERS              = 3.
      IF sy-subrc <> 0.
        CONTINUE.
      ENDIF.
      APPEND INITIAL LINE TO li_dep ASSIGNING FIELD-SYMBOL(<fs_dep>).
      MOVE 'PROG'          TO <fs_dep>-obj_type.
      MOVE lw_include_name TO <fs_dep>-object_name.
    ELSEIF ls_e071-object = 'FUNC'.
      MOVE lw_objname TO lw_fm_name.
      CALL FUNCTION 'FUNCTION_INCLUDE_INFO'
*       IMPORTING
*         FUNCTAB                   =
*         NAMESPACE                 =
*         PNAME                     =
        CHANGING
          funcname            = lw_fm_name
*         GROUP               =
          include             = lw_fm_include
        EXCEPTIONS
          function_not_exists = 1
          include_not_exists  = 2
          group_not_exists    = 3
          no_selections       = 4
          no_function_include = 5
          OTHERS              = 6.
      IF sy-subrc <> 0.
        CONTINUE.
      ENDIF.
      APPEND INITIAL LINE TO li_dep ASSIGNING FIELD-SYMBOL(<fs_dep2>).
      MOVE 'PROG'        TO <fs_dep2>-obj_type.
      MOVE lw_fm_include TO <fs_dep2>-object_name.
    ELSEIF ls_e071-object = 'REPS'.
      APPEND INITIAL LINE TO li_dep ASSIGNING FIELD-SYMBOL(<fs_dep3>).
      MOVE 'PROG'     TO <fs_dep3>-obj_type.
      MOVE lw_objname TO <fs_dep3>-object_name.
    ELSEIF ls_e071-object = 'CPUB'.
      MOVE lw_objname TO lw_class_name.
      DATA(lw_pub_include) = cl_oo_classname_service=>get_pubsec_name( lw_class_name ).
      APPEND INITIAL LINE TO li_dep ASSIGNING FIELD-SYMBOL(<fs_dep4>).
      MOVE 'PROG'     TO <fs_dep4>-obj_type.
      MOVE lw_pub_include TO <fs_dep4>-object_name.

    ELSEIF ls_e071-object = 'CPRI'.
      MOVE lw_objname TO lw_class_name.
      DATA(lw_pri_include) = cl_oo_classname_service=>get_prosec_name( lw_class_name ).
      APPEND INITIAL LINE TO li_dep ASSIGNING FIELD-SYMBOL(<fs_dep5>).
      MOVE 'PROG'     TO <fs_dep5>-obj_type.
      MOVE lw_pri_include TO <fs_dep5>-object_name.

    ELSEIF ls_e071-object = 'CPRO'.
      MOVE lw_objname TO lw_class_name.
      DATA(lw_pro_include) = cl_oo_classname_service=>get_prisec_name( lw_class_name ).
      APPEND INITIAL LINE TO li_dep ASSIGNING FIELD-SYMBOL(<fs_dep6>).
      MOVE 'PROG'     TO <fs_dep6>-obj_type.
      MOVE lw_pro_include TO <fs_dep6>-object_name.

    ELSEIF ls_e071-object = 'CLSD'.
      MOVE lw_objname TO lw_class_name.
      DATA(lw_def_include) = cl_oo_classname_service=>get_ccdef_name( lw_class_name ).
      APPEND INITIAL LINE TO li_dep ASSIGNING FIELD-SYMBOL(<fs_dep7>).
      MOVE 'PROG'     TO <fs_dep7>-obj_type.
      MOVE lw_def_include TO <fs_dep7>-object_name.
    ELSE.
      APPEND INITIAL LINE TO li_dep ASSIGNING FIELD-SYMBOL(<fs_dep1>).
      MOVE ls_e071-object TO <fs_dep1>-obj_type.
      MOVE lw_objname     TO <fs_dep1>-object_name.
    ENDIF.

    CALL FUNCTION 'ZCR_GET_DEPENDENCY_OBJ'
      EXPORTING
        ii_objects = li_dep
      IMPORTING
        ev_json    = lw_json.

    CLEAR li_dep.


* Generate a line in the output
    ls_output-line = |=========================================|.
    APPEND ls_output TO li_output.
    ls_output-line = |Object: { ls_e071-obj_name } { ls_e071-object }|.
    APPEND ls_output TO li_output.
    ls_output-line = |Dependency JSON: { lw_json }|.
    APPEND ls_output TO li_output.
    CLEAR lw_json.

    IF ls_e071-object = 'PROG'.
      ls_e071-object = 'REPS'.
    ENDIF.

* Get version list for the object
    CALL FUNCTION 'SVRS_GET_VERSION_DIRECTORY_46'
      EXPORTING
        objname      = lw_objname
        objtype      = ls_e071-object
      TABLES
        lversno_list = li_versions
        version_list = li_version_list
      EXCEPTIONS
        OTHERS       = 1.
    IF sy-subrc IS NOT INITIAL.
      CONTINUE.
    ENDIF.


*  Here We are discarding TOC from the version List
    li_trkorr = VALUE #( FOR ls_ver IN li_version_list
                       ( ls_ver-korrnum ) ).

    DELETE ADJACENT DUPLICATES FROM li_trkorr.

    SELECT trkorr,
           trfunction
      INTO TABLE @DATA(li_e070)
      FROM e070
      FOR ALL ENTRIES IN @li_trkorr
     WHERE trkorr = @li_trkorr-table_line.

    LOOP AT li_version_list ASSIGNING FIELD-SYMBOL(<fs_version>).
      READ TABLE li_e070 INTO DATA(ls_e070)
           WITH KEY trkorr = <fs_version>-korrnum.

      IF sy-subrc NE 0 OR ls_e070-trfunction EQ 'T'.
        CLEAR <fs_version>-korrnum.
      ENDIF.

    ENDLOOP.
* Version order is like this
* 000 -> active
* 003 -> just before
* 002 -> second
* 001 -> first
* So, the comparison is to take place between 000 and 003
* But in some cases the Version request number of 000 is empty which means that 000 and 003 are identical
* For that case, we compare 003 and 002, so we delete all versions where KORRNUM is empty

    DELETE li_version_list WHERE korrnum IS INITIAL.

    IF lines( li_version_list ) < 2.
      ls_output-line = |This is the first version of this object.|.
      APPEND ls_output TO li_output.
      ls_output-line = | |.
      APPEND ls_output TO li_output.
      lw_only_one_version = abap_true.
    ENDIF.

    IF rb_full IS NOT INITIAL.
      lw_only_one_version = abap_true.
    ENDIF.

    READ TABLE li_version_list INTO ls_latest_ver INDEX 1.
    IF sy-subrc IS NOT INITIAL.
      CONTINUE.
    ENDIF.

* Get source code for latest version
    CALL FUNCTION 'SVRS_GET_REPS_FROM_OBJECT'
      EXPORTING
        object_name = lw_objname
        object_type = ls_e071-object
        versno      = ls_latest_ver-versno
      TABLES
        repos_tab   = li_code_new
        trdir_tab   = li_trdir_new
      EXCEPTIONS
        OTHERS      = 1.
    IF sy-subrc IS NOT INITIAL.
      CONTINUE.
    ENDIF.

    IF lw_only_one_version = abap_false.
      READ TABLE li_version_list INTO ls_prev_ver INDEX 2.
      IF sy-subrc IS NOT INITIAL.
        CONTINUE.
      ENDIF.

* Get source code for previous version
      CALL FUNCTION 'SVRS_GET_REPS_FROM_OBJECT'
        EXPORTING
          object_name = lw_objname
          object_type = ls_e071-object
          versno      = ls_prev_ver-versno
        TABLES
          repos_tab   = li_code_old
          trdir_tab   = li_trdir_old
        EXCEPTIONS
          OTHERS      = 1.
      IF sy-subrc IS NOT INITIAL.
        CONTINUE.
      ENDIF.

* Get delta between the two versions
      CALL FUNCTION 'SVRS_COMPUTE_DELTA_REPS'
        TABLES
          texttab_old  = li_code_old
          texttab_new  = li_code_new
          trdirtab_old = li_trdir_old
          trdirtab_new = li_trdir_new
          trdir_delta  = li_trdir_delta
          text_delta   = li_delta
        EXCEPTIONS
          OTHERS       = 1.
      IF sy-subrc IS NOT INITIAL.
        CONTINUE.
      ENDIF.
    ELSE.
      LOOP AT li_code_new ASSIGNING FIELD-SYMBOL(<fs_code_new>).
        APPEND INITIAL LINE TO li_delta ASSIGNING FIELD-SYMBOL(<fs_delta>).
        <fs_delta>-line = <fs_code_new>-line.
        <fs_delta>-number = sy-tabix.
        <fs_delta>-vrsflag = 'I'.
      ENDLOOP.

    ENDIF.

* For differential method
    IF rb_diff IS NOT INITIAL OR rb_full IS NOT INITIAL.
* For each line in the delta, generate a new line in output file
      LOOP AT li_delta INTO ls_delta.

        CASE ls_delta-vrsflag(1).
          WHEN 'I'.         "Insertion
            ls_output-line = |LINE { sy-tabix }: + { ls_delta-line }|.
            APPEND ls_output TO li_output.

          WHEN 'D'.         "Deletion
            ls_output-line = |LINE { sy-tabix }: - { ls_delta-line }|.
            APPEND ls_output TO li_output.

          WHEN 'M' OR 'U'. "Modified
            ls_output-line = |LINE { sy-tabix }: ~ { ls_delta-line }|.
            APPEND ls_output TO li_output.

          WHEN OTHERS.
            CONTINUE.
        ENDCASE.

      ENDLOOP.
    ENDIF.

* For blocks method
    IF rb_bloc IS NOT INITIAL.
* ---------------------------------------------------------------------*
* 1. Reconstruct a unified code list containing unchanged & changed lines
* ---------------------------------------------------------------------*
      CLEAR li_unified_code.
      lw_idx_old = 1.
      lw_idx_new = 1.

      LOOP AT li_delta INTO ls_delta.
        CASE ls_delta-vrsflag(1).
          WHEN 'D'. " Deletion
            WHILE lw_idx_old < ls_delta-number AND lw_idx_old <= lines( li_code_old ).
              READ TABLE li_code_old INTO ls_code_old INDEX lw_idx_old.
              ls_unified-flag = ' '.
              ls_unified-line = ls_code_old-line.
              APPEND ls_unified TO li_unified_code.
              lw_idx_old = lw_idx_old + 1.
              lw_idx_new = lw_idx_new + 1.
            ENDWHILE.

            ls_unified-flag = 'D'.
            ls_unified-line = ls_delta-line.
            APPEND ls_unified TO li_unified_code.
            lw_idx_old = lw_idx_old + 1.

          WHEN 'I'. " Insertion
            WHILE lw_idx_new < ls_delta-number AND lw_idx_new <= lines( li_code_new ).
              READ TABLE li_code_new INTO ls_code_new INDEX lw_idx_new.
              ls_unified-flag = ' '.
              ls_unified-line = ls_code_new-line.
              APPEND ls_unified TO li_unified_code.
              lw_idx_new = lw_idx_new + 1.
              lw_idx_old = lw_idx_old + 1.
            ENDWHILE.

            ls_unified-flag = 'I'.
            ls_unified-line = ls_delta-line.
            APPEND ls_unified TO li_unified_code.
            lw_idx_new = lw_idx_new + 1.

          WHEN 'M' OR 'U'. " Modified/Updated
            WHILE lw_idx_new < ls_delta-number AND lw_idx_new <= lines( li_code_new ).
              READ TABLE li_code_new INTO ls_code_new INDEX lw_idx_new.
              ls_unified-flag = ' '.
              ls_unified-line = ls_code_new-line.
              APPEND ls_unified TO li_unified_code.
              lw_idx_new = lw_idx_new + 1.
              lw_idx_old = lw_idx_old + 1.
            ENDWHILE.

            ls_unified-flag = '~'.
            ls_unified-line = ls_delta-line.
            APPEND ls_unified TO li_unified_code.
            lw_idx_new = lw_idx_new + 1.
            lw_idx_old = lw_idx_old + 1.
        ENDCASE.
      ENDLOOP.

      " Catch any remaining unchanged lines at the end of the file
      WHILE lw_idx_new <= lines( li_code_new ).
        READ TABLE li_code_new INTO ls_code_new INDEX lw_idx_new.
        ls_unified-flag = ' '.
        ls_unified-line = ls_code_new-line.
        APPEND ls_unified TO li_unified_code.
        lw_idx_new = lw_idx_new + 1.
      ENDWHILE.

* ---------------------------------------------------------------------*
* 2. Parse the unified stream to extract changed blocks
* ---------------------------------------------------------------------*
      CLEAR: li_block_buffer, lw_in_block, lw_block_changed.

      LOOP AT li_unified_code INTO ls_unified.

        lw_text = ls_unified-line.
        CONDENSE lw_text.

        " Extract the first keyword, ignoring full-line comments
        CLEAR lw_w1.
        IF lw_text IS NOT INITIAL AND lw_text(1) <> '*' AND lw_text(1) <> '"'.
          SPLIT lw_text AT space INTO lw_w1 lw_w2.
          TRANSLATE lw_w1 TO UPPER CASE.
          REPLACE ALL OCCURRENCES OF '.' IN lw_w1 WITH space.
          CONDENSE lw_w1.
        ENDIF.

        " Start of a measurable block?
        IF lw_in_block = abap_false AND
           ( lw_w1 = 'FORM' OR lw_w1 = 'METHOD' OR lw_w1 = 'FUNCTION' OR lw_w1 = 'MODULE' OR lw_w1 = 'DEFINE' ).
          lw_in_block = abap_true.
          CLEAR: li_block_buffer, lw_block_changed.

          lw_block_name = lw_w2.
          REPLACE ALL OCCURRENCES OF '.' IN lw_block_name WITH space.
          CONDENSE lw_block_name.

          ls_output-line = |--- Block: { lw_w1 } { lw_block_name } ---|.

          DATA(lw_temp_tabix) = sy-tabix.
          APPEND ls_output TO li_block_buffer.
          sy-tabix = lw_temp_tabix.
          CLEAR lw_temp_tabix.
        ENDIF.

        " Format the output line based on flag
        CLEAR ls_output.
        CASE ls_unified-flag.
          WHEN 'I'.
            ls_output-line = |LINE { sy-tabix }: + { ls_unified-line }|.
            lw_block_changed = abap_true.
          WHEN 'D'.
            ls_output-line = |LINE { sy-tabix }: - { ls_unified-line }|.
            lw_block_changed = abap_true.
          WHEN '~'.
            ls_output-line = |LINE { sy-tabix }: ~ { ls_unified-line }|.
            lw_block_changed = abap_true.
          WHEN OTHERS.
            ls_output-line = |LINE { sy-tabix }: = { ls_unified-line }|.
        ENDCASE.

        IF lw_in_block = abap_true.
          " Inside a block: accumulate all lines (changed and unchanged)
          APPEND ls_output TO li_block_buffer.

          " End of the block?
          IF lw_w1 = 'ENDFORM' OR lw_w1 = 'ENDMETHOD' OR lw_w1 = 'ENDFUNCTION' OR lw_w1 = 'ENDMODULE' OR lw_w1 = 'END-OF-DEFINITION'.
            lw_in_block = abap_false.

            " Only push block to file if it was flagged as dirty
            IF lw_block_changed = abap_true.
              APPEND LINES OF li_block_buffer TO li_output.
              ls_output-line = | |.
              APPEND ls_output TO li_output.
            ENDIF.
            CLEAR: li_block_buffer, lw_block_changed.
          ENDIF.

        ELSE.
          " Outside a block (e.g. global data declarations)
          " Push changes directly to file
          IF ls_unified-flag <> ' '.
            APPEND ls_output TO li_output.
          ENDIF.
        ENDIF.

      ENDLOOP.

      " Flush buffer in case the block didn't properly terminate
      IF lw_in_block = abap_true AND lw_block_changed = abap_true.
        APPEND LINES OF li_block_buffer TO li_output.
        ls_output-line = | |.
        APPEND ls_output TO li_output.
      ENDIF.
    ENDIF.

    ls_output-line = | |.
    APPEND ls_output TO li_output.

  ENDLOOP.

  CALL FUNCTION 'GUI_DOWNLOAD'
    EXPORTING
      filename         = lw_full_path
      filetype         = 'ASC'
    TABLES
      data_tab         = li_output
    EXCEPTIONS
      file_write_error = 1
      OTHERS           = 2.

  IF sy-subrc = 0.
    WRITE: / 'File successfully downloaded to:', lw_full_path.
  ELSE.
    WRITE: / 'Error downloading file. Check if the file path is valid.'.
  ENDIF.
